\chapter{Data Generating Function}

The idea of a data generating function comes from \cite{Lindqvist2025simulated}.

\section{Definition and Examples}

\begin{definition}[Data Generating Function]
	Let $T$ be a random variable whose distribution depends on an unknown
	parameter $\theta$. A \textbf{Data Generating Function} (DGF) for $T$ is a
	function $\tau$ together with a random vector $\mathbf U$ whose distribution
	is known and does not depend on $\theta$, such that for every possible
	value of $\theta$
	\[
	T \overset{d}{=} \tau(\mathbf U, \theta).
	\]
\end{definition}

The random vector $\mathbf{U}$ must not depend on $\theta$. Thus, the randomness and the parameter $\theta$ are separated.
Meaning the DGF is a transformation of a random vector $\mathbf U$ using $\theta$, such that 
\[
\mathbf{U} \overset{\tau(\cdot, \theta)}{\longrightarrow} T.
\]

Intuitively, one can say that the data $T$ comes from randomness $\mathbf U$ and the parameter $\theta$.
To understand the concept of DGF, we will look at the Examples~\ref{example: DGF normal}--\ref{example: non invertible DGF}.

\begin{example}[DGF for Normal Distribution] \label{example: DGF normal}
	Let $T \sim \mathcal{N}(\theta, 1)$. Then we can write
	\[
	T \sim \theta + U,
	\]
	where $U \sim \mathcal{N}(0, 1)$. Thus, $U$ does not depend on the parameter $\theta$. We can define the DGF as
	\begin{equation*}
		\tau(U, \theta) = \theta + U.
	\end{equation*}
	
	Given $\theta$, the DGF $\tau$ generates simulated data for $T$. Now, assume that we observe $T = t$ and that $U = u$ is given. We can then obtain an estimate of the parameter $\theta$ by inverting $\tau$.
	The inverse is given by
	\[
	\hat{\theta}(u, t) = t - u.
	\]
	
	In this example, the inversion works perfectly because $\tau$, viewed as a function of $\theta$, is bijective and therefore invertible. However, there are also DGFs that cannot be inverted, as illustrated in Example \ref{example: non invertible DGF}.
\end{example}


\begin{example}[DGF with $n > 1$] \label{example:dgf with n large}
	We can generalize Example \ref{example: DGF normal} by defining
	\[
	T = \sum_{i=1}^n X_i,
	\]
	where $X_1, \ldots, X_n \overset{\text{iid}}{\sim} \mathcal{N}(\theta, 1)$.
	As in Example \ref{example: DGF normal}, we can write
	$X_i = \theta + U_i$ for $i \in \{1, \ldots, n\}$, where
	$U_1, \ldots, U_n \overset{\text{iid}}{\sim} \mathcal{N}(0, 1)$.
	It follows that
	\[
	T = \sum_{i=1}^n (\theta + U_i)
	= n\theta + \sum_{i=1}^n U_i.
	\]
	Thus, the DGF is defined as
	\begin{align} \label{dgf of multi normal}
		\tau(\mathbf{U}, \theta) = n\theta + \sum_{i=1}^n U_i.
	\end{align}
	
	This DGF is also invertible, with inverse
	\begin{align} \label{dgf of multi normal fiducial distn}
		\hat{\theta}(\mathbf{u}, t) =
		\frac{t - \sum_{i=1}^n u_i}{n}.
	\end{align}
	
	Figure~\ref{fig:dgf example} shows the DGF
	$\tau(\mathbf{U}, \cdot)$ for a fixed value of $\mathbf{U}$,
	viewed as a function of $\theta$. The DGF is a straight line
	with slope $n$ and intercept $\sum_{i=1}^n U_i$.
	Since $\tau$ is a linear function in $\theta$,
	we can easily solve the equation
	$\tau(\mathbf{U}, \theta) = t$ for $\theta$, obtaining
	$\hat{\theta}(\mathbf{U}, t)$, as illustrated in
	Figure~\ref{fig:dgf example inverse}.

\begin{example}[Non invertible DGF] \label{example: non invertible DGF}
	
	Let \(T \sim \text{Bin}(n,\theta)\) with success probability $\theta \in \Theta = [0, 1]$.
	
	Let \(\mathbf U=(U_1,\ldots,U_n)\) be a random vector whose components are independent and uniformly distributed on \([0,1]\). For each \(i \in \{1, \ldots , n\}\),  it holds
	$U_i \leq \theta$ if and only if
	$\mathds{1}(U_i\leq\theta)=1$
	and therefore since \(U_i\) is uniformly distributed on \([0,1]\),
	\[
	\mathbb P\left(\mathds{1}(U_i \leq \theta)=1\right)
	=
	\mathbb P(U_i \leq \theta)
	=
	\theta
	\]
	for all $i \in \{1, \ldots , n\}$.
	Thus,
	\[
	\mathds{1}(U_i \leq \theta)\sim\operatorname{Bin}(1,\theta)
	\]
	for all \(i\in\{1,\ldots,n\}\). Since the random variables $U_1, \ldots , U_n$ are independent and therefore also their transformations $\mathds{1}(U_1 \leq \theta), \ldots ,  \mathds{1}(U_n \leq \theta)$, their sum is binomially distributed too. Hence,
	\[
	\sum_{i=1}^n \underbrace{\mathds{1}(U_i \leq \theta)}_{\sim \text{Bin}(1, \theta)}
	\sim \operatorname{Bin}(n,\theta).
	\]
	
	Using this result, we define the DGF by
	\begin{align} \label{dgf in clopper pearson case}
		\tau(\mathbf U,\theta)
		=
		\sum_{i=1}^n \mathds{1}(U_i\leq\theta)
	\end{align}
	and by construction it holds
	\(
	T\overset{d}{=}\tau(\mathbf U,\theta)
	\).	
	To illustrate the DGF, let us fix \(\mathbf U\) and consider \(\tau(\mathbf U,\theta)\) as a function of \(\theta\).
	
	For example, let \(n=2\), and let \(U_{1:2}\) and \(U_{2:2}\) denote the order statistics of \(\mathbf U\), that is,
	\[
	U_{1:2}=\min(U_1,U_2)
	, \quad
	U_{2:2}=\max(U_1,U_2).
	\]
	Then the DGF is
	\begin{align}
		\tau (\mathbf U, \theta)
		&=
		\sum_{i = 1}^2 \mathds{1} (U_i \leq \theta)
		=
		\mathds{1} (U_1 \leq \theta) +  \mathds{1} (U_2 \leq \theta)
		\notag \\ 
		&= 
		\mathds{1} (U_{1:2} \leq \theta) +  \mathds{1} (U_{2:2} \leq \theta) 
		=
		\begin{cases}
			0, \text{ if } \theta \in [0, U_{1:2}] \\
			1, \text{ if } \theta \in (U_{1:2}, U_{2:2}] \\
			2, \text{ if } \theta \in (U_{2:2}, 1]
		\end{cases}
		\label{dgf for clopper pearson example}
	\end{align}
	
	Figure~\ref{fig:clopper_pearson_interval_dgf} shows this function schematically.
	
	For example if we observe $t = 1$, then $\hat \theta (\mathbf U, 1) \in (U_{1:2}, U_{2:2}]$, so there is no unique solution. For any observation $t \in \{0, 1, 2\}$, we have infinitely many solutions for the parameter $\hat \theta (\mathbf U, t)$. In particular, the DGF $\tau$ is not invertible.
	\begin{figure}[h]
		\centering
		\includegraphics[width=\textwidth]{figures/lindqvist/clopper_pearson_interval_dgf.pdf}
		\caption{DGF $\tau(\mathbf U , \theta) = n \theta + \sum_{i = 1}^n U_i$ for fixed $\mathbf U$ and $n = 2$ (see \eqref{dgf for clopper pearson example})}
		\label{fig:clopper_pearson_interval_dgf}
	\end{figure}

\end{example}




\section{Construction of CI using DGF}

\subsection{Constructing a CB}

\cite{Lindqvist2025simulated} presents Theorem~1 in his paper, which provides a method for constructing an upper CB for the parameter $\theta$ based on a DGF.
\begin{theorem}\label{theorem:dgf-upper bound}
	Let $T$ be a random variable with parameter $\theta \in \Theta$ and let $\tau$ be the corresponding DGF.
	Define \[\bar \theta ( \mathbf u, t) = \sup \{ \theta: \tau (\mathbf u, \theta) \leq t \}\] and let $a(t)$ be a left-continuous function such that
	\begin{align} \label{precondition theorem 1}
		\mathbb{P} \left( \bar \theta (\mathbf U, t)  \geq a(t)\right) \leq \alpha
	\end{align}
	for all $t$ and $\alpha \in (0, 1)$. 
	Then $a(T)$ is an upper $(1-\alpha)$-CB for $\theta$, that is 
	\[
	\mathbb{P} \left( \theta \geq a(T) \right) \leq \alpha 
	\]
	for all $\alpha \in (0, 1)$.
\end{theorem}


This construction of a CB does not require the DGF \(\tau\) to be monotone or continuous, allowing the method to also address all DGFs.
To illustrate the construction of the mapping $a$, we consider the following example.

\begin{example}[Construction of $a$ from Theorem~\ref{theorem:dgf-upper bound}]\label{example:easy to find a}
	Suppose $T \sim \text{Exp} \left( \frac{1}{\theta} \right)$ and $U \sim \text{Exp}(1)$. Then we can express $T$ in terms of the function $\tau$ if we define it as 
	\[ \tau (U , \theta ) = \theta \cdot U\]
	because then 
	\begin{equation} \label{exp distn rescale}
		\mathbb{P} (\tau (U , \theta ) \leq x) = 
		\mathbb{P} \left( \theta \cdot U \leq x \right) =
		\mathbb{P} \left( U \leq \frac{x}{\theta} \right) = 
		1 - \exp \left( - \frac{x}{\theta} \right)
	\end{equation}
	which is the CDF of an exponentially distributed random variable with parameter $ \frac{1}{\theta}$. So it holds $T \overset{d}{=} \tau (U, \theta)$. Now we calculate $\bar \theta (u, t)$ from Theorem~\ref{theorem:dgf-upper bound} as
	\[
		\bar \theta (u, t) = 
		\sup \left\{ \theta : \tau (u, \theta) \leq t \right\}
		=
		\sup \left\{ \theta : \theta \cdot u \leq t \right\}
		=
		\sup \left\{ \theta : \theta \leq \frac{t}{u} \right\}
		=
		\frac{t}{u}.
	\]
	Let $\alpha \in (0, 1)$ be arbitrary.
	The next objective is to find the function $a$ such that~\eqref{precondition theorem 1} holds for all $\alpha \in (0, 1)$. We can also find $a$ such that \eqref{precondition theorem 1} holds with equality by calculating
	\begin{align} \label{theorem 1 finding 1 in example}
		\alpha 
		\overset{!}{=}
		\mathbb{P} \left( \bar \theta (U, t) \geq a(t) \right)
		=
		\mathbb{P} \left( \frac{t}{U} \geq a(t) \right)
		=
		1 - \exp \left( - \frac{t}{a(t)} \right).
	\end{align}
	We can solve \eqref{theorem 1 finding 1 in example} for $a(t)$ and get
	\[
		a(t) = \frac{- t}{\ln(1- \alpha)}.
	\]
	This is a left-continuous function because it is a linear function of \(t\). Therefore, the preconditions for Theorem~\ref{theorem:dgf-upper bound} are satisfied and we can apply it.
	After observing $T$, one can say that 
	\[
		\left( 0, a(T) \right] = \left( 0,  \frac{-T}{\ln(1- \alpha)} \right]
	\]
	is a one-sided $(1-\alpha)$-CI for $\theta$. For example for $\alpha = 0.1$ the upper $(1-\alpha)$-confidence bound is given by $\frac{-T}{\ln(1 - 0.1)} \approx 9.49 T$.
\end{example}


In Example~\ref{example:easy to find a}, it was straightforward to construct the mapping \(a\) required for Theorem~\ref{theorem:dgf-upper bound}. In general, however, finding an appropriate mapping \(a\) can be difficult, as Example~\ref{example:hard to find a} illustrates.

\begin{example}\label{example:hard to find a}
	Assume that $T$ is a random variable with DGF
	\[
	\tau (\mathbf U, \theta) := \theta U_1 + \theta^2 U_2,
	\]
	where $U_1, U_2$ are independent and exponentially distributed with rate $1$, and $\theta \in (0, \infty)$.
	Then
	\[
	\bar \theta (\mathbf u, t)
	=
	\sup \{ \theta : \theta u_1 + \theta^2 u_2 \leq t \}
	=
	\frac{- u_1 + \sqrt{u_1^2 + 4 u_2 t}}{2 u_2}.
	\]
	
	Analogously to \eqref{theorem 1 finding 1 in example}, we try to find the mapping $a$ such that
	\begin{align}
		\alpha
		&\overset{!}{=}
		\mathbb{P} \left( \bar \theta (\mathbf U, t) \geq a(t) \right) \notag \\
		&=
		\mathbb{P} \left(
		\frac{- U_1 + \sqrt{U_1^2 + 4 U_2 t}}{2 U_2}
		\geq a(t) \right). \label{right hand side theorem i beispiel}
	\end{align}
	In Example~\ref{example:easy to find a}, the distribution of the term inside the probability in \eqref{right hand side theorem i beispiel} was known, but in this case, the distribution of this term is not easy to determine.
\end{example}

A caveat of finding an upper CB with Theorem~\ref{theorem:dgf-upper bound} is that the function $a$ must be found.

On the other hand, in contrast to the CIs defined in Section~\ref{section:sundberg}, this CB is always valid, i.e., it attains the nominal level for every $\alpha \in (0,1)$ without asymptotic arguments. In particular, if $T$ depends on a sample size $n$, the CB from Theorem~\ref{theorem:dgf-upper bound} also holds for small $n$. If $a$ is chosen such that \eqref{precondition theorem 1} holds with equality, the coverage is often exactly $1-\alpha$, as in Example~\ref{example:easy to find a}.

Next, we proof Theorem~\ref{theorem:dgf-upper bound}.
	
\begin{proof}
	Let $\alpha \in (0, 1)$ be arbitrary but fixed. Let $\bar \theta$ be defined as in Theorem~\ref{theorem:dgf-upper bound}, and let $a$ be a left-continuous function such that
	\begin{align*}
		\mathbb{P} \left( \bar \theta (\mathbf U, t) \geq a(t) \right) \leq \alpha
	\end{align*}
	for all $t$.
	We need to show that $a(T)$ is an upper CB, i.\,e.\
	\begin{align} \label{to show theorem 1}
		\mathbb{P} \left( \theta \geq a(T) \right) \leq \alpha
	\end{align}
	for all $\theta \in \Theta$.
	
	Let us define
	\[
	a^{-1} (\theta ) = \sup \{ t : a(t) \leq \theta \}.
	\]
	
	First, we show five statements in step i) to step v). Then, we use these statements in step vi) to prove the inequality \eqref{to show theorem 1}.
	
	\textbf{Step (i):} \\	
	Assume $\theta \geq a(T)$. Then $T \in \{ t: a(t) \leq \theta \}$. Since $T$ is an element of this set, $T$ has to be smaller or equal to the supremum of this set, i.e. 
	\[
		T \leq \sup \{ t: a(t) \leq \theta \} = a^{-1}(\theta).
	\]
	This is a pure consequence of the definition of the supremum of a set. All in all, the inequality $\theta \geq a(T)$, implies $T \leq a^{-1}(T)$. 
	So we can conclude that
	\[
		\{ \theta \geq a(T) \} \subseteq \{ T \leq a^{-1}(\theta) \}.
	\]
	and therefore 
	\begin{align} \label{theorem 1: step i}
		\mathbb{P} \left( \theta \geq a(T) \right) \leq 
		\mathbb{P} \left( T \leq a^{-1} (\theta) \right).
	\end{align}	
	
	
	\textbf{Step (ii):} \\
	
	Since $T \overset{d}{=} \tau (\mathbf U, \theta)$ because $\tau$ is a DGF of $T$, we can say that the equality 
	\begin{align} \label{theorem 1: step ii}
		\mathbb{P} \left( T \leq a^{-1}(\theta) \right)
		=
		\mathbb{P} \left( \tau (\mathbf U, \theta) \leq a^{-1}(\theta) \right)
	\end{align}
	holds.
	
	\textbf{Step (iii):} \\
	Assume $ \tau ( \mathbf U, \theta ) \leq a^{-1}(\theta)$. Define $t' := a^{-1}(\theta)$, then it holds $ \tau ( \mathbf U, \theta ) \leq t'$. So, we can observe that $\theta \in \{\theta' : \tau(\mathbf U, \theta') \leq t'\}$. Again, with the properties of the supremum, we get
	\[
		\theta \leq \sup \{\theta' : \tau(\mathbf U, \theta') \leq t'\}
		= \bar \theta (\mathbf U, t').
	\]
	All in all, it holds that $\tau (\mathbf U, \theta) \leq t'$ implies that $\theta \leq \bar \theta (\mathbf U, t')$. We can use this to know that 
	\[
			\left\{ 
				\tau (\mathbf U, \theta) \leq a^{-1} (\theta) 
			\right\}
		\subseteq
			\left\{ 
				\bar \theta \left( \mathbf U, a^{-1}(\theta) \right)  \geq \theta
			\right\}.
	\]
	and obtain the inequality
	\begin{align} \label{theorem 1: step iii}
		\mathbb{P} \left( \tau (\mathbf U, \theta) \leq a^{-1} (\theta) \right)
		\leq 
		\mathbb{P} \left( \bar \theta \left( \mathbf U, a^{-1}(\theta) \right)  \geq \theta \right).
	\end{align}
	

	
	
	\textbf{Step (iv):} \\
	We want to show that $a\big(a^{-1}(\theta)\big) \le \theta$.
	This is where left-continuity of $a$ is used. Let
	\[
	t^\ast := a^{-1}(\theta) = \sup\{t : a(t) \le \theta\}.
	\]
	If the defining set $\{t : a(t) \le \theta\}$ has a maximum, i.e. contains $t^\ast$ itself, then trivially $a(t^\ast) \le \theta$ and we are done.
	
	Otherwise, $t^\ast$ is only a limit point of the set. There is a sequence $t_n \nearrow t^\ast$ with $a(t_n) \le \theta$ for every $n$. Left-continuity of $a$ means
	\[
	a(t_n) \longrightarrow a(t^\ast) \quad
	\]
	as $t_n \nearrow t^\ast$.
	Since $a(t_n) \le \theta$ for all $n$, and the limit of a sequence bounded above by $\theta$ is itself smaller or equal to$\theta$ and we obtain
	\(
	a(t^\ast) \le \theta,
	\)
	so
	\(
	a\big(a^{-1}(\theta)\big) \le \theta.
	\)
	So we can conclude that 
	\begin{align} \label{theorem 1: step iv}
		\mathbb{P} \left( \bar \theta (\mathbf U, a^{-1}(\theta)) \geq \theta \right)
		\geq 
		\mathbb{P} \left( \bar \theta (\mathbf U, a^{-1}(\theta)) \geq a \left( a^{-1} (\theta) \right) \right).
	\end{align}
	
	
	\textbf{Step (v):} \\
	As a last step, we show a direct instance of the hypothesis~\eqref{precondition theorem 1}
	applied with the particular choice $t = a^{-1}(\theta)$.
	We get 
	\begin{align} \label{theorem 1: step v}
		\mathbb{P}_U\Big( \theta\big(U, a^{-1}(\theta)\big) \ge a\big(a^{-1}(\theta)\big) \Big) \le \alpha.
	\end{align}
	
	\textbf{Step (vi):} \\
	In the end, we can combine the knowledge derived in step (i) to step (v) to show that $a(T)$ is an upper CB. It holds
	\begin{align*}
		\mathbb{P} \left( \theta \geq a(T) \right)
		&\overset{\eqref{theorem 1: step i}}{\leq}
		\mathbb{P} \left( T \leq a^{-1}(\theta) \right) \\
		&\overset{\eqref{theorem 1: step ii}}{=}
		\mathbb{P} \left( \tau(\mathbf U, \theta) \leq a^{-1}(\theta) \right) \\
		&\overset{\eqref{theorem 1: step iii}}{\leq}
		\mathbb{P} \left[
		\bar \theta (\mathbf U, a^{-1}(\theta)) \geq \theta
		\right] \\
		&\overset{\eqref{theorem 1: step iv}}{\leq}
		\mathbb{P} \left[ 
		\bar \theta (\mathbf U, a^{-1}(\theta)) \geq a^{-1}(a(\theta))
		\right] \\
		&\overset{\eqref{theorem 1: step v}}{\leq}
		\alpha
	\end{align*}
	and this shows that $a(T)$ is an upper CB with confidence level $1 - \alpha$.
	
\end{proof}


The following corollary presents a corresponding result to Theorem~\ref{theorem:dgf-upper bound}, but in this case, we aim to construct a lower CB.
The proof works in the same way as the proof of Theorem~\ref{theorem:dgf-upper bound}.


\begin{corollary} \label{corollary:dgf-lower bound}
	Let $T$ be a random variable with parameter $\theta \in \Theta$ and let $\tau$ be the corresponding DGF. Define
	\[
		\underline \theta (\mathbf u, t) = \inf \{\theta: \tau(\mathbf u, \theta) \geq t\}
	\]
	and let $b(t)$ be a right-continuous function such that
	\[
	\mathbb{P} \left( \underline \theta (\mathbf U, t)  \leq b(t)\right) \leq \alpha
	\]
	for all $t$ and $\alpha \in (0, 1)$. 
	Then $b(T)$ is a lower CB for $\theta$ that is $\mathbb{P} \left( \theta \leq b(T) \right) \leq \alpha $ for all $\alpha \in (0, 1)$.
\end{corollary}

\begin{proof}
	The proof of Corollary~\ref{corollary:dgf-lower bound} is analogous to the proof of Theorem~\ref{theorem:dgf-upper bound}, but we define
	\[
	b^{-1}(\theta) = \inf \{ t: b(t) \geq \theta \}
	\]
	instead of $a^{-1}$. Steps i) to v) work analogously, except that the direction of the inequality signs have to be reversed.
\end{proof}


\subsection{Constructing a CI}

Under the additional assumption that the DGF is strictly increasing in $\theta$,
the values of $\underline \theta (\mathbf u, t)$ and $\bar \theta (\mathbf u, t)$ from Theorem~\ref{theorem:dgf-upper bound} and
Corollary~\ref{corollary:dgf-lower bound} are the same, and the confidence
niveau holds with equality, what is summarized in the following corollary.

\begin{corollary} \label{corollary:Corollary 2 Lindqvist}
	Let $\tau(\mathbf u, \theta)$ be strictly increasing as a function of $\theta$. 
	Then 
	\[
		\underline \theta (\mathbf u, t) = \bar \theta (\mathbf u, t)
	\]
	for all $\mathbf u, t$. Also if $ \bar \theta (\mathbf U, t)$ has a continuous distribution and $a(t)$ and $b(t)$ are chosen with equality in Theorem~\ref{theorem:dgf-upper bound} and Corollary~\ref{corollary:dgf-lower bound}, then 
	\[
		\mathbb{P} \left( \theta \geq a(T) \right)
		=
		\mathbb{P} \left( \theta \leq b(T) \right)
		=
		\alpha.
	\]
\end{corollary}


\begin{proof}
	Let $\tau$ be strictly increasing as a function of $\theta$.
	
	In the first part, we show that $\underline \theta (\mathbf u, t) = \bar \theta (\mathbf u, t)$.
	
	For easier notation, denote 
	\[
		A := \{ \theta: \tau (\mathbf U, \theta) \leq t \} 
		\quad \text{and} \quad
		B := \{ \theta: \tau (\mathbf U, \theta) \geq t \}.
	\]
	Note that per construction $\underline \theta (\mathbf u, t) = \sup (A)$ and $\bar \theta (\mathbf u, t) = \inf (B)$. 
	So the goal is to show that $\sup (A) = \inf (B)$.
	
	It holds $A \cup B = \Theta$ since for all $\theta \in \Theta$, it holds $\tau (\mathbf U, \theta) \leq t$ or $\tau (\mathbf U, \theta) \geq t$. 
	Let $a \in A$ and $b \in B$, then $\tau (\mathbf U, a) \leq t \leq \tau (\mathbf U, b)$. Since $\tau$ is strictly increasing in $\theta$, it must holds $a \leq b$. $a$ and $b$ were arbitrary, so it holds $a \leq b$ for all $a \in A, b \in B$. Therefore $\sup (A) \leq \inf (B)$. To show that equality holds, we assume that $\sup (A) < \inf (B)$ and lead it to a contradiction. Assume $\sup (A) < \inf (B)$. Then there must be a $\theta' \in \Theta$ with $\sup (A) < \theta' < \inf (B)$. But this means that $\theta' \notin A$ and $\theta' \notin B$ what is a contradiction to $A \cup B = \Theta$. 
	So we can follow that $\sup (A) = \inf (B)$ which is equivalent with $\underline \theta (\mathbf u, t) = \bar \theta (\mathbf u, t)$.
	
	The second part of the proof shows the exactness, i. e. $\mathbb{P} \left( \theta \geq a(T) \right) = \mathbb{P} \left( \theta \leq b(T) \right) = \alpha$. 
	We first concentrate on $a$. The proof for $b$ is similar.
	We will split the proof into seven steps.
	
	\textbf{Step (i):} Introduce $F^\vartheta$
	
	Denote 
	\[
	F^{\vartheta}(t) = \mathbb{P} \left( \tau (\mathbf U, \vartheta) \leq t \right)
	\]
	which is a CDF, so $F^{\vartheta}$ is right-continuous in $t$. Also $F^{\vartheta}$ is non-increasing in $\vartheta$. 
	To see this, let $\vartheta_1 < \vartheta_2$. 
	Since $\tau$ is strictly increasing in $\vartheta$, it holds 
	\[
	\tau(\mathbf u, \vartheta_1) \leq \tau(\mathbf u, \vartheta_2), 
	\]
	so for all $t$ holds
	\[
	\left\{ \tau(\mathbf u, \vartheta_2) \leq t \right\} \subseteq 
	\left\{ \tau(\mathbf u, \vartheta_1) \leq t \right\}
	\]
	implying that
	\[
	\mathbb{P} \left( \tau(\mathbf u, \vartheta_2) \leq t \right)
	\leq 
	\mathbb{P} \left( \tau(\mathbf u, \vartheta_1) \leq t \right).
	\]
	Using the notation of $F^{\vartheta}$ gives for $\vartheta_1 < \vartheta_2$  
	\[
	F^{\vartheta_2} (t) \leq F^{\vartheta_1} (t)
	\]
	what concludes that $F^{\vartheta}$ is non-increasing in $\vartheta$.
	Note that $F^\theta$ is the CDF of $T$, so $\mathbb{P} (T \leq t)  = F^\theta (t)$.
	
		
	\textbf{Step (ii):} Show $F^{a(t)} (t) = \alpha$ for all $t$ \\
	
	Let $t$ be fix and $\vartheta = a(t)$.
	\begin{itemize}
		\item 
		Show $\bar \theta (\mathbf u, t) > \vartheta \implies t \geq \tau (\mathbf u, \vartheta)$ \\
		Let $\bar \theta (\mathbf u, t) > \vartheta$. Then $\theta' \in A$ exists with $\theta' > \vartheta$. 
		Since $\tau$ is scrictly increasing in $\theta$, it holds 
		\[
		\tau (\mathbf u, \theta')
		>
		\tau (\mathbf u, \vartheta)
		\]
		Since $\theta' \in A$, it also holds $t \geq \tau (\mathbf u, \theta')$ resulting in 
		\[
		t \geq \tau (\mathbf u, \vartheta).
		\]
		\item 
		Show $\tau (\mathbf u, \vartheta) \leq t \implies \vartheta \leq \bar \theta (\mathbf u, t)$ \\
		Let $\tau (\mathbf u, \vartheta) \leq t$, then according to the definition of $A$, it holds $\vartheta \in A$, so $\vartheta$ is smaller than the supremum of the set $A$
		\[
		\vartheta \leq \sup(A) = \bar \theta (\mathbf u, t).
		\]
	\end{itemize}
	We can summarize these results as
	\[
	\left\{ 
	\bar \theta (\mathbf u, t) > \vartheta
	\right\}
	\subseteq
	\left\{ 
	\tau (\mathbf u, \vartheta) \leq t
	\right\}
	\subseteq
	\left\{ 
	\bar \theta (\mathbf u, t) \geq \vartheta
	\right\}.
	\]
	We can observe the probabilities of these events and notice that 
	\[
	\mathbb{P}
	\left(
	\bar \theta (\mathbf u, t) > \vartheta
	\right) 
	\leq
	\mathbb{P}
	\left(
	\tau (\mathbf u, \vartheta) \leq t
	\right) 
	\leq
	\mathbb{P}
	\left(
	\bar \theta (\mathbf u, t) \geq \vartheta
	\right) 
	.
	\]
	The middle term is equal to $F^{\vartheta}(t)$ and regarding to the precondition on $a(t) = \vartheta$ it holds 
	$\mathbb{P}
	\left(
	\bar \theta (\mathbf u, t) \geq a(t)
	\right) = \alpha$ and since $\bar \theta(\mathbf u, t)$ has a continuous distribution, also $\mathbb{P}
	\left(
	\bar \theta (\mathbf u, t) > \vartheta
	\right)  = \alpha$. So we can conclude that	for all $t$
	\[
	F^{a(t)} (t) 
	=
	\alpha .
	\]
	
	\textbf{Step (iii): Introduce notation and split the probability} \\
	
	Define 
	\[
		S := \left\{ t : a(t) \leq \theta \right\}.
	\]
	Let $t$ and $\theta \in \Theta$ be fix. Then there are two cases.
	\begin{enumerate}
		\item 
		$t \in S$: 
		Then it holds $a(t) \leq \theta$ what implies $F^{a(t)}(t) \geq F^\theta(t)$ since $F^\theta$ is non-increasing in $\theta$. 
		In step (ii), we showed that $F^{a(t)}(t) = \alpha$, so it holds $\alpha \geq F^\theta(t)$.
		\item 
		$t \notin S$: In this case, we have $a(t) > \theta$, what implies $\alpha \leq F^\theta(t)$.
	\end{enumerate}
	Those cases are equivalent to 
	\begin{align}
		&F^\theta (t) > \alpha \implies t \notin S \label{cor 2 case 1} 
		\intertext{and}
		&F^\theta (t) < \alpha \implies t \in S \label{cor 2 case 2}.
	\end{align}
	
	Define 
	\begin{align*}
		q_1 := \inf \left\{ t : F^{\theta}(t) \geq \alpha \right\}
		\quad \text{and} \quad
		q_2 := \inf \left\{ t : F^{\theta}(t) > \alpha \right\}.
	\end{align*}
	It holds $\left\{  F^{\theta}(t) \geq \alpha \right\} \supseteq \left\{  F^{\theta}(t) > \alpha \right\}$, so $q_1 \leq q_2$.
	For proving the exactness, we need to show $\mathbb{P} \left( a(T) \leq \theta \right) = \alpha$ but the left side of this equation is the same as $\mathbb{P} \left( T \in S \right)$. We can split $S$ into three parts
	\begin{align*}
		S = 
		\underbrace{\left\{
			t \in (- \infty, q_1] : a(t) \leq \theta
		\right\}}_{ = S_1}
		\cup
		\underbrace{\left\{
		t \in (q_1, q_2) : a(t) \leq \theta
		\right\}}_{ = S_2}
		\cup
		\underbrace{\left\{
		t \in [q_2, \infty) : a(t) \leq \theta
		\right\}}_{ = S_3}.
	\end{align*}
	Since $S_1, S_2$ and $S_3$ are disjoint sets, we can write
	\begin{align} \label{cor 2 aufteilung S}
		\mathbb{P} \left( T\in S \right)
		=
		\mathbb{P} \left( T\in S_1 \right)
		+
		\mathbb{P} \left( T\in S_2 \right)
		+
		\mathbb{P} \left( T\in S_3 \right)
	\end{align}
	and calculate each probability seperately.
	
	
	\textbf{Step (iv):} Calculate $\mathbb{P} \left( T \in S_1 \right)$
	
	
	First, we show that $T \in (- \infty , q_1]$ implies $a(T) \leq \theta$. 
	Let $T \in (- \infty , q_1]$. We can observe two cases.
	\begin{enumerate}
		\item 
		$T < q_1$: \\
		By the definition of the infimum, it must hold $T \notin \{t : F^\theta(t) \geq \alpha\}$. Otherwise $T$ would be the infimum of this set, not $q_1$. We can follow, that $F^\theta(T) < \alpha$ and with \eqref{cor 2 case 2}, we get $T \in S$, what means that $a(T) \leq \theta$.
		\item 
		$T = q_1$: \\
		Define the sequence $(t_n)_{n \in \mathbb N}$, with $t_n = q_1 - 1 / n$ for all $n \in \mathbb N$. Then $t_n \nearrow q_1$. For all $n \in \mathbb N$, holds $t_n < q_1$, so with the first case we can conclude for all $n \in \mathbb N$ that $a(t_n) \leq \theta$. Since $a$ is a left-continuous function, we also get that 
		\begin{equation} \label{cor 2 a(q_1)}
			a(q_1) \leq \theta.
		\end{equation}
	\end{enumerate}
	So we can say that in general
	\begin{equation} \label{cor 2 sets}
		\left\{ T \leq q_1 \right\}
		\subseteq
		\left\{ a(T) \leq \theta \right\}.
	\end{equation}
	Intersecting both sides of~\eqref{cor 2 sets} with the set $\left\{ T \leq q_1 \right\}$ and considering that the left set is a subset of the right gives  
	\begin{equation} \label{cor 2 mengen}
		\left\{ T \leq q_1 \right\}
		=
		\underbrace{\left\{ a(T) \leq \theta \right\} \cap \left\{ T \leq q_1 \right\}}_{ = \{T \in S_1\}}.
	\end{equation}
	
	We can use all this knowledge to calculate the probability
	\begin{align} \label{cor 2 vorbereitung S_1}
		\mathbb{P} \left( T \in S_1 \right)
		\overset{\eqref{cor 2 mengen}}{=}
		\mathbb{P} \left( T \leq q_1 \right)
		=
		F^\theta (q_1). 
	\end{align}
	$q_1$ is constructed such that $F^\theta (q_1) \geq \alpha$. Assume it holds $F^\theta (q_1) > \alpha$. Then it follows with~\eqref{cor 2 case 1}, that $q_1 \notin S$ what implies $a(q_1) > \theta$. This is a contradiction to~\eqref{cor 2 a(q_1)}. So the assumption $F^\theta (q_1) > \alpha$ is false. In conclusion, it holds 
	\begin{equation}  \label{cor 2 S_1}
		\mathbb{P} \left( T \in S_1 \right) 
		\overset{\eqref{cor 2 vorbereitung S_1}}{=}
		F^\theta (q_1)
		=
		\alpha.
	\end{equation}
	
	
	\textbf{Step (v):} Calculate $\mathbb{P} \left( T \in S_2 \right)$ \\
	
	We know that $S_2 \subseteq (q_1, q_2)$, so we can conclude that 
	\begin{equation} \label{cor 2 step iv prob}
		\mathbb{P} \left( T \in S_2 \right)
		\leq 
		\mathbb{P} \left( q_1 < T < q_2 \right)
		=
		\lim_{t \nearrow q_2} F^\theta (t) - F^\theta (q_1).
	\end{equation}
	We can observe that $F^\theta$ is constant on $[q_1, q_2)$ because if $t \in [q_1, q_2)$, then $t \geq q_1$ what implies that $F^\theta (t) \geq F^\theta (q_1)$ since $F^\theta$ is monotone increasing in $t$. With the use of \eqref{cor 2 S_1}, we can bound $F^\theta (t) \geq \alpha$. But it also holds that $t < q_2$ and with the definition of the infimum, it must hold $F^\theta (t) \leq \alpha$. Combining these results gives 
	\begin{equation} \label{cor 2 step iv F}
		F^\theta (t) = \alpha \text{ for all } t \in [q_1, q_2).
	\end{equation}
	We can follow that $\lim_{t \nearrow q_2} F^\theta (t) = \alpha$ and therefore we can use \eqref{cor 2 step iv prob} to obtain 
	\begin{equation*}
		\mathbb{P} \left( T \in S_2 \right)
		\overset{\eqref{cor 2 step iv prob}}{\leq}
		\lim_{t \nearrow q_2} F^\theta (t) - F^\theta (q_1)
		\overset{\eqref{cor 2 S_1} + \eqref{cor 2 step iv F}}{=}
		\alpha - \alpha
		=
		0.
	\end{equation*}
	what finally gives us 
	\begin{equation} \label{cor 2 S_2}
		\mathbb{P} \left( T \in S_2 \right) = 0.
	\end{equation}
	
	\textbf{Step (vi):} Calculate $\mathbb{P} \left( T \in S_3 \right)$ \\
	
	In the following, we will show that $t > q_2$ implies that $t \notin S$. Since $F^\theta$ is monotone increasing, we know that for $t > q_2$
	\begin{equation*}
		F^\theta (t) \geq F^\theta (q_2).
	\end{equation*}
	Since $q_2$ is the infimum of the set $\{t: F^\theta (t) > \alpha\}$, so there must be a $s \in \{t: F^\theta (t) > \alpha\}$ with $q_2 > s$. Also with the fact that $F^\theta$ is monotone increasing, we can follow 
	\begin{align*}
		F^\theta (q_2) \geq F^\theta (s)
	\end{align*}
	and since $s$ is in the set, it holds $F^\theta (s) > \alpha$. Combining these observations gives 
	\begin{equation*}
		F^\theta (t) \geq F^\theta (q_2) \geq F^\theta (s) > \alpha.
	\end{equation*}
	So $F^\theta (t) > \alpha$ holds.
	With \eqref{cor 2 case 1}, we can follow that $t \notin S$. 
	Therefore $S_3 \subseteq \{q_2\}$. It holds $F^\theta (q_2) \geq \alpha$ because $F^\theta$ is monotone increasing and \eqref{cor 2 step iv F}. We can distinguish between two cases.
	\begin{enumerate}
		\item 
		$F^\theta (q_2) > \alpha$: \\
		Then with \eqref{cor 2 case 1} follows $q_2 \notin S$ and per construction of $S_3$ also $q_2 \notin S_3$. Therefore $S_3 = \varnothing$ and $\mathbb{P} \left( T \in S_3 \right) = 0$.
		\item 
		$F^\theta (q_2) = \alpha$: \\
		Then 
		\begin{equation*}
			\mathbb{P} (T \in S_3) = \mathbb{P} (T = q_2)
			=
			F^\theta (q_2) - \lim_{t \nearrow q_2} F^\theta (t).
		\end{equation*}
		It holds $F^\theta (q_2) = \alpha$ per construction and with \eqref{cor 2 step iv F}, we can follow that also in this case
		\begin{equation*}
			\mathbb{P} (T \in S_3)
			=
			F^\theta (q_2) - \lim_{t \nearrow q_2} F^\theta (t)
			=
			\alpha - \alpha = 0.
		\end{equation*}
	\end{enumerate}
	So for all possible cases holds
	\begin{equation} \label{cor 2 S_3}
		\mathbb{P} (T \in S_3) = 0.
	\end{equation}
	
	\textbf{Step (vii):} Calculate $\mathbb{P} (T \in S)$ \\
	
	We split the probability $\mathbb{P} (T \in S)$ into three parts in \eqref{cor 2 aufteilung S} and calculated every part what gives us
	\begin{align*}
		\mathbb{P} \left( \theta \geq a(T) \right)
		=&
		\mathbb{P} (T \in S)  \\
		\overset{\eqref{cor 2 aufteilung S}}{=} &
		\mathbb{P} (T \in S_1) + \mathbb{P} (T \in S_2) + \mathbb{P} (T \in S_3) \\
		\overset{\text{Step (iv) - (vi)}}{=} &
		\alpha + 0 + 0
		=
		\alpha.
	\end{align*}
	This proves the exactness for the function $a$. We can show $\mathbb{P} \left( \theta \leq b(T) \right) = \alpha$ analogously. In Table~\ref{tab:comparison a b} in the Appendix, all major differences between the proofs of the exactness are summarized.
\end{proof}	


\subsection{Example: Binomial Distribution}

The example from this section is discussed by~\cite{Lindqvist2025simulated} in Section 2.1.
\textcolor{red}{($\star$) erwähne dass es Clopper Pearson Intervals sind }

\textcolor{red}{($\star$) start, ab hier generall über n nicht direkt in $n = 2$ starten}


Let \(T \sim \mathrm{Bin}(n,\theta)\), and let \(t \in \{0,\ldots,n\}\) be the observed value of \(T\). The goal is to construct a CI for the success probability \(\theta \in \Theta=[0,1]\).

We already analysed this situation in Example~\ref{example: non invertible DGF}, so we know that the function given in \eqref{dgf in clopper pearson case} is a valid DGF of $T$. 

To use Theorem~\ref{theorem:dgf-upper bound}, we define
\[
\bar \theta (\mathbf u, t) 
=
\sup \{ \theta : \tau(\mathbf u, \theta) \leq t \}
=
\sup \left\{ \theta : \sum_{i = 1}^n \mathds{1}(u_{i:n} < \theta) \leq t \right\}
=
\begin{cases}
	u_{1:n} &\text{ for } t = 0, \\
	u_{2:n} &\text{ for } t = 1, \\
	\vdots & \\
	u_{(n-1):n} &\text{ for } t = n-2, \\
	u_{n:n} &\text{ for } t = n-1, \\
	1 &\text{ for } t = n, \\
\end{cases}
\]
All in all, $\bar \theta (\mathbf u, t) = u_{(t+1):n}$ for all $t \in \{0, \ldots , n\}$ whereas we define $u_{(n+1):n} := 1$.



Next, we want to find a left-continuous function $a$ such that $\mathbb{P} \left( \bar \theta (\mathbf U, t) \geq a(t) \right) \leq \alpha$ for all $t\in \{0, \ldots , n\}$ and $\alpha \in (0, 1)$.
To construct $a(t)$, we consider all possible values of $t$ and choose $a(t)$ accordingly. Assume that $t\in\{0,\ldots,n-1\}$. We require that
\begin{align}
	\alpha
	\overset{!}{=}
	\mathbb{P}\left(
	\bar{\theta}(\mathbf{U},t) \geq a(t)
	\right) 
	=
	\mathbb{P}\left(
	U_{(t+1):n} \geq a(t)
	\right).
	\label{criteria clopper pearson}
\end{align}
The order statistics of independent standard uniformly distributed random variables follow Beta distributions, i.e.,
\[
U_{k:n}\sim\operatorname{Beta}(k,n-k+1),
\]
for all $k \in \{1, \ldots, n\}$. This is proven in Example 2.3. in \cite{david2004order}.
We can use the quantiles of the Beta distribution to determine $a(t)$ for all $t \in \{0, \ldots, n\}$.
Let $\beta_q (v, w)$ denote the $q$-quantile of the $\text{Beta}(v, w)$ distribution. 
If we define
\[
a(t)=\beta_{1-\alpha}(t+1,n-t),
\]
this choice of $a(t)$ satisfies the criterion in~\eqref{criteria clopper pearson} for all $t \in \{0, \ldots , n-1\}$.

For $t = n$ we have $\bar\theta(\mathbf U, n) = 1$, which is deterministic, so
$\mathbb P(\bar\theta(\mathbf U,n) \geq a(n)) \in \{0,1\}$ and equality with $\alpha$ is impossible.
Instead, we can directly choose $a(n)= 1$ because if we observe $T = n$, then it is natural to get the upper CB of $1$.

All in all, define the mapping $a$ as

\begin{equation} \label{a for clopper pearson}
	a(t) = 
	\begin{cases}
		\beta_{1- \alpha}(t+1, n-t) &\text{for } t < n, \\
		1 &\text{for } t = n.
	\end{cases}
\end{equation}

If we observe $T = t$, we can conclude with Theorem~\ref{theorem:dgf-upper bound} that $[0, a(T)]$ whereas $a$ is defined as in \eqref{a for clopper pearson}, is an upper $(1-\alpha)$-CI for $\theta$.

For further insight, let us consider the case $\alpha = 0.1$ and \(n=2\). In this case, \(\bar{\theta}\) is given by
\[
\bar \theta(\mathbf{u},t)
=
\begin{cases}
	u_{1:2}, & \text{if } t=0,\\
	u_{2:2}, & \text{if } t=1,\\
	1, & \text{if } t=2.
\end{cases}
\]
The function \(a\) is given by
\[
a(t)
=
\begin{cases}
	\beta_{1-\alpha}(1,2), & \text{if } t=0,\\
	\beta_{1-\alpha}(2,1), & \text{if } t=1,\\
	1, & \text{if } t=2.
\end{cases}
\]

Tables~\ref{tab:upper-CB-example-0.9} and~\ref{tab:upper-CB-example-0.99} show the resulting upper CIs for all possible observations of $T$. 

Next, we construct a lower \((1-\alpha)\)-CI based on Corollary~\ref{corollary:dgf-lower bound}. First, we observe that
\begin{align}
	\label{precondition for corollary 1}
	\underline{\theta}(\mathbf U,t)
	&= U_{(t:n)}.
\end{align}
As in the construction of \(a\), we can find a function \(b\). In particular,
\[
b(t)=
\begin{cases}
	0, & \text{if } t=0,\\
	\beta_{\alpha}(t,n-t+1), & \text{if } t>0.
\end{cases}
\]
By Corollary~\ref{corollary:dgf-lower bound}, for an observed value \(T=t\), the interval \([b(t),1]\) is a lower \((1-\alpha)\)-CI for \(\theta\). Concrete examples of these intervals are given in Tables~\ref{tab:lower-CB-example-0.9} and~\ref{tab:lower-CB-example-0.99}.

	\frac{R}{TTT} = \frac{
	\sum_{i = 1}^n \mathds{1}(X_i \leq c)
	}{
	\sum_{i = 1}^n \min(X_i, c)}
\end{align}
which matches the MLE of $\theta$ as we derived in~\eqref{MLE for type I}.
To seperate the randomness and the parameter $\theta$ which is now not seperated since $X_i$ is a random variable dependent on the parameter $\theta$, we can use a trick. Define $U_1, \ldots , U_n $ as independent standard exponentially distributed random variables. 
We already saw in \eqref{exp distn rescale}, that $U_i / \theta$ is exponentially distributed with parameter $\theta$. So $X_i \overset{d}{=} U_i / \theta$ and we can substitute this into \eqref{T for exp distn}. 
So if we define 
\begin{align}
	\tau(\mathbf U, \theta)
	= 
	\frac{
	\sum_{i = 1}^n \mathds{1}\left(\frac{U_i}{\theta} \leq c \right)
	}{
	\sum_{i = 1}^n \min \left(\frac{U_i}{\theta}, c \right)}
\end{align}
it holds $T \overset{d}{=} \tau(\mathbf U, \theta)$, so $\tau$ is a DGF for the MLE.


Since $U_i / \theta < c$ if and only if $U_i / c < \theta$ for all $i \in \{1, \ldots , n\}$, and the ordering of the summands does not matter, we can use the order statistics
\begin{align} 	\label{DGF for MLE of Exp distn}
	\tau(\mathbf U, \theta) = \frac
	{
		\sum_{i = 1}^n  \mathds{1} \left( \frac{U_{i:n}}{c} < \theta \right)
	}{
		\sum_{i = 1}^n \min \left(  \frac{U_{i:n}}{\theta} , c \right)
	}
\end{align}


To illustrate the DGF \eqref{DGF for MLE of Exp distn}, we first consider the simple case $n = 2$. In this case, the DGF takes the form
\begin{align} \label{dgf for n = 2}
	\tau (\mathbf U, \theta)
	=
	\frac
	{
		\mathds{1} \left( \frac{U_{1:2}}{c} < \theta \right)
		+
		\mathds{1} \left( \frac{U_{2:2}}{c} < \theta \right)
	}{
		\min \left(  \frac{U_{1:2}}{\theta} , c \right)
		+
		\min \left(  \frac{U_{2:2}}{\theta} , c \right)
	}. 	
\end{align}



Since the numerator can only take the values $0$, $1$, and $2$, we consider three cases.


\textbf{Case A: $R = 0$}

If $R=0$, the numerator of \eqref{dgf for n = 2} is equal to zero. This means that neither condition $A_1$ nor condition $A_2$ is satisfied, where
\[
A_1 := \left\{\frac{U_{1:2}}{c}<\theta\right\},
\qquad
A_2 := \left\{\frac{U_{2:2}}{c}<\theta\right\}.
\]

Therefore, it holds that 
$\frac{U_{1:2}}{c}\geq\theta$
and
$\frac{U_{2:2}}{c}\geq\theta$.
Since
\(
\frac{U_{1:2}}{c}\leq\frac{U_{2:2}}{c}
\)
and $\theta>0$, these two conditions can be simplified to
\[
\theta\in\left(0,\frac{U_{1:2}}{c}\right].
\]

Since the numerator of \eqref{dgf for n = 2} is zero, the whole fraction is zero. Hence,
\[
\tau(\mathbf U,\theta)=0
\]
for all
$\theta\in\left(0,\frac{U_{1:2}}{c}\right]$.


\textbf{Case B: $R = 1$}

If $R=1$, the numerator of \eqref{dgf for n = 2} is equal to one. Therefore, exactly one of the conditions $A_1$ and $A_2$ is satisfied.

Since
\[
\frac{U_{2:2}}{c}<\theta
\quad\Longrightarrow\quad
\frac{U_{1:2}}{c}<\theta,
\]
because $U_{1:2}\leq U_{2:2}$ by the definition of order statistics, condition $A_2$ implies condition $A_1$. Therefore, if exactly one condition is satisfied, $A_1$ is satisfied but $A_2$ is not. Thus,
\begin{equation} \label{inequalitites R = 1}
	\frac{U_{1:2}}{c}<\theta
	\qquad\text{and}\qquad
	\frac{U_{2:2}}{c}\geq\theta,
\end{equation}
which means that
\[
\theta\in
\left(
\frac{U_{1:2}}{c},
\frac{U_{2:2}}{c}
\right].
\]

Hence,
$\mathds{1}(A_1)=1$
and
$\mathds{1}(A_2)=0$ and the numerator is equal to $1$.

We can also determine the two minimums in the denominator of\eqref{dgf for n = 2} with the inequalities~\eqref{inequalitites R = 1}
\[
\min\left(\frac{U_{1:2}}{\theta},c\right)
=
\frac{U_{1:2}}{\theta}
\]
and
\[
\min\left(\frac{U_{2:2}}{\theta},c\right)
=
c.
\]

Thus, for
$\theta\in
\left(
\frac{U_{1:2}}{c},
\frac{U_{2:2}}{c}
\right]$,
the DGF is given by 
\begin{align}
	\tau(\mathbf U,\theta)
	=
	\frac{1 + 0}{
		\frac{U_{1:2}}{\theta}+c
	}
	=
	\frac{\theta}{U_{1:2}+c\theta}.
\end{align}


\textbf{Case C: $R = 2$}

If $R=2$, both indicators are equal to one. Therefore, both conditions $A_1$ and $A_2$ are satisfied. In particular,
\[
\frac{U_{2:2}}{c}<\theta,
\]
which implies
\[
\theta\in
\left(
\frac{U_{2:2}}{c},
\infty
\right).
\]

From the inequalities of $A_1$ and $A_2$ follows that the minimums in the denominator are
\[
\min\left(\frac{U_{i:2}}{\theta},c\right)
=
\frac{U_{i:2}}{\theta}
\]
for $i \in \{1, 2\}$.

Thus, for $\theta\in \left( \frac{U_{2:2}}{c}, \infty \right)$, the DGF is given by
\[
\tau(\mathbf U,\theta)
=
\frac{1+1}{
	\frac{U_{1:2}}{\theta}
	+
	\frac{U_{2:2}}{\theta}
}
=
\frac{2\theta}{U_{1:2}+U_{2:2}}.
\]


Combining all three cases A - C gives
\begin{align} \label{dgf for exp with cases}
\tau(\mathbf U,\theta)
=
\left\{
\begin{array}{lll}
	0,
	&
	\text{if }
	\theta\in
	\left(
	0,\frac{U_{1:2}}{c}
	\right],
	&
	\qquad (A),
	\\[8pt]
	\displaystyle
	\frac{\theta}{U_{1:2}+c\theta},
	&
	\text{if }
	\theta\in
	\left(
	\frac{U_{1:2}}{c},
	\frac{U_{2:2}}{c}
	\right],
	&
	\qquad (B),
	\\[10pt]
	\displaystyle
	\frac{2\theta}{U_{1:2}+U_{2:2}},
	&
	\text{if }
	\theta\in
	\left(
	\frac{U_{2:2}}{c},
	\infty
	\right),
	&
	\qquad (C).
\end{array}
\right.
\end{align}


For a fixed $\mathbf U = (U_1, U_2)$, we can plot $\tau (\mathbf U, \theta)$ as a function of $\theta$. This is illustrated in Figure~\ref{fig:Figure 1 from Lindqvist}. We can see that this is a piecewise monotonically increasing function, with jumps at $\frac{U_{1:2}}{c}$ and $\frac{U_{2:2}}{c}$.

can be solved with $\theta \in \left( 0, \frac{u_{1:2}}{c} \right]$. So there is no unique solution.

\section{Construction of CI using DGF and Monte Carlo}

\cite{Lindqvist2025simulated} describe with Theorem~2 and Corollary~3 and 4 methods to find CBs and a CI using a Monte Carlo simulation. The advantage over the methods presented in Theorem~\ref{theorem:dgf-upper bound} and Corollary~\ref{corollary:dgf-lower bound} and~\ref{corollary:Corollary 2 Lindqvist} is that we do not need to know the distribution of $\bar \theta$ and $\underline \theta$ and do not need to find the functions $a$ and $b$.



\begin{theorem}
	Let the.... \textcolor{red}{($\star$) preconditions}.
	Let $\mathbf U_1, \ldots , \mathbf U_m$ be independent with the same distirbution as $\mathbf U$. 
	
	Let $\bar \theta_{1:m}, \ldots , \bar \theta_{m:m}$ be the ordered values of $\bar \theta(U_j, t)$ for $j \in \{1, \ldots , m\}$ and 
	$\underline \theta_{1:m}, \ldots , \underline \theta_{m:m}$ be the ordered values of $\underline \theta(U_j, t)$ for $j \in \{1, \ldots , m\}$.
	Let $k \in \{1 , \ldots , \lfloor \frac{m+1}{2} \rfloor\}$.
	Then the following statements hold.
	\begin{enumerate}[label=(\alph*)]
		\item \label{theorem_monte_carlo_a}
		$\bar \theta_{(m-k+1):m}$ is a (possibly conservative) exact upper $\left(1 - \frac{k}{m+1} \right)$-CB for $\theta$.
		\item \label{theorem_monte_carlo_b}
		$\underline \theta_{k:m}$ is a (possibly conservative) exact lower $\left(1 - \frac{k}{m+1} \right)$-CB for $\theta$
		\item \label{theorem_monte_carlo_c}
		If $\tau(\mathbf u, \theta)$ is strictly increasing in $\theta$, then 
		\[
			\left[ \bar \theta_{k:m}, \bar \theta_{(m-k+1):m} \right]
		\]
		is an exact $\left( 1 - \frac{2k}{m+1} \right)$-CI.
	\end{enumerate}
\end{theorem}


\begin{proof}% [of \ref{theorem_monte_carlo_a}]
	Firstly, we show \ref{theorem_monte_carlo_a}.
	
	The goal is to show 
	\[
		\mathbb{P} \left( \theta \leq \bar \theta_{(m-k+1):m} (\mathbf u, T) \right) \geq 1 - \frac{k}{m+1}.
	\]
	We split the proof into five steps.
	
	\textbf{Step (i):} Show $\theta > \bar \theta (\mathbf u, T) \implies \tau(\mathbf u, \theta) > T$
	
	Let $\theta > \bar \theta (\mathbf u, T)$. 
	Suppose it holds 
	$\tau(\mathbf u, \theta) \leq T$. Then 
	\[
		\theta \in \{ \vartheta \in \Theta : \tau (\mathbf u, \vartheta) \leq T \}
	\]
	and since $\theta$ is an element of this set, it is also samller than its supremum which is exactly $\bar \theta (\mathbf u, T)$
	\[
		\theta
		\leq
		\sup \{ \vartheta \in \Theta : \tau (\mathbf u, \vartheta) \leq T \} = \bar \theta (\mathbf u, T).
	\]
	This is a contradiction to the assumption $\theta > \bar \theta (\mathbf u, T)$, so it must hold that $\tau(\mathbf u, \theta) > T$.
	
	\textbf{Step (ii):} Introduce Notation
	
	The probability $\mathbb{P} \left( \theta > \bar \theta_{(m-k+1):m} (\mathbf u, T) \right)$ depends on the random vectors $\mathbf U_1, \ldots , \mathbf U_m$ and $T$. If we set $\mathbf U = \mathbf U_0$, it holds 
	$T = \tau (\mathbf U, \theta)$. Define 
	\[
		R_j := \tau (\mathbf U_j, \theta)
	\] 
	for all $j \in \{0, \ldots , m\}$. Note that $R_0 = T$ holds. All of the random variables $R_0, \ldots , R_m$ are iid because the underlying random variables $\mathbf U_0, \ldots , \mathbf U_m$ are iid.
	
	For the further proof, we will use the notation
	\[
		N_i := \left| \{ j \in \{0, \ldots , m\} \backslash \{i\}  : R_j \leq R_i \} \right|
	\]
	what counts the number of $R_j$ which are smaller or equal to $R_i$.
	
	\textbf{Step (iii):} Show $\theta > \bar \theta _{(m-k+1):m} \implies N_0 \leq k-1$
	
	Let $\theta > \bar \theta _{(m-k+1):m}$. Then there are at least $m-k+1$ indices $j \in \{1, \ldots , m\}$ for which holds
	\[
		\theta > \bar \theta_j.
	\]
	Let $j_1, \ldots , j_{m-k+1}$ be these indices. From step (i), we know that we can conclude
	\[
		\tau (\mathbf U_j, \theta) > T
	\]
	for all $j \in \{j_1, \ldots , j_{m-k+1}\}$ and using the definition of $R_j$, we get
	\[
		R_j > R_0
	\]
	for all $j \in \{j_1, \ldots , j_{m-k+1}\}$. Since there are at least $m-k+1$ indices for which $R_j > R_0$, there are at most $k-1$ indices for which $R_j \leq R_0$. 
	This means that 
	\[
		N_0 
		=
		\left| \{ j \in \{1, \ldots , m\}   : R_j \leq R_0 \} \right| 
		\leq k-1.
	\]
	In conclusion, we showed that if $\theta > \bar \theta _{(m-k+1):m}$ holds then $N_0 \leq k-1$ also holds. Therefore we can write 
	\begin{equation} \label{rank proof uff 3}
		\mathbb{P} \left(\theta > \bar \theta _{(m-k+1):m}  \right)
			\leq 
		\mathbb{P} \left(N_0 \leq k-1  \right).
	\end{equation}
	
	
	\textbf{Step (iv):} bound $\mathbb{P} (N_0 \leq k-1)$
	
	For this step, we first observe that 
	\begin{equation} \label{rank proof uff 1}
		\sum_{i = 0}^m \mathds{1} (N_i \leq k-1) \leq k 
	\end{equation}
	because at most $k$ of the $m+1$ values can be among the $k$ smallest. If $k+1$ indices had $N_i \leq k-1$, then the one with the largest $R_i$ among them would have at least $k$ other smaller values itself which is a contradiction.
	
	Also the random variables $R_0, \ldots , R_m$ are iid, so the probability $\mathbb{P} (N_i \leq k-1)$ is the same for all $i \in \{0, \ldots , m\}$. So 
	\begin{equation} \label{rank proof uff}
		\sum_{i = 0}^m \mathbb{P} (N_i \leq k-1) = (m+1) \mathbb{P} (N_0 \leq k-1).
	\end{equation} 
	The left side of \eqref{rank proof uff} is equal to $\mathbb E \left[ \sum_{i = 0}^m \mathds{1} (N_i \leq k-1) \right]$, which has to be bounded by k regarding to \eqref{rank proof uff 1}. We can combine 
	
	\begin{align}
		\mathbb{P} (N_0 \leq k-1) 
		\overset{\eqref{rank proof uff}}{=} &
		\frac{1}{m+1} \cdot \sum_{i = 0}^m \mathbb{P} (N_i \leq k-1)
		\notag
		\\
		=&
		\frac{1}{m+1} \cdot \mathbb E \left[ \sum_{i = 0}^m \mathds{1} (N_i \leq k-1) \right]
		\notag
		\\
		\overset{\eqref{rank proof uff 1}}{\leq}&
		\frac{k}{m+1}. \label{rank proof uff 4}
	\end{align}
	
	\textbf{Step (v): Combine the results}
	
	Combining the results of Steps (iii) and (iv) yields
	\begin{equation*}
		\mathbb{P} \left( \theta > \bar \theta_{(m-k+1):m} \right)
		\overset{\eqref{rank proof uff 3}}{\leq}
		\mathbb{P} \left( N_0 \leq k-1 \right)
		\overset{\eqref{rank proof uff 4}}{\leq}
		\frac{k}{m+1}.
	\end{equation*}
	Taking the complement, we obtain
	\[
	\mathbb{P} \left( \theta \leq \bar \theta_{(m-k+1):m} \right) \geq 1 - \frac{k}{m+1}.
	\]
	Hence $\bar \theta_{(m-k+1):m}$ is an upper CB for $\theta$ with confidence level $1 - \frac{k}{m+1}$.
\end{proof}

\subsection{Example: Hybrid Type-I-censored Exponentially Distributed Lifetimes}

In Sections~\ref{section:sundberg} and \ref{section:Type I Censored Exponentially Distributed Lifetimes}, we consider type-I-censored exponential lifetimes and construct CIs for the MLE of the rate parameter \(\theta\in\Theta=(0,\infty)\).

In this section, we will use a hybrid censoring scheme instead. Just as in Section~\ref{section:Type I Censored Exponentially Distributed Lifetimes}, we will construct the DGF for the MLE $\hat \theta$.


Assume $X_1, \ldots, X_n \overset{\text{iid}}{\sim} \text{Exp}(\theta)$. We want to find the MLE of the parameter $\theta \in \Theta = (0, \infty)$ based on the censored data $Y_1, \ldots, Y_n$. 


The experiment stops at the time
\[
C_{\text{stop}} = \min \left( X_{r:n}, c \right),
\]
that is, as soon as either the $r$-th failure has occurred or the time $c$ has been reached, whichever happens first. 
This gives two cases, depending on whether $R \geq r$ or $R < r$.

Recall that each observation contributes to the likelihood as follows
\[
f_{Y_i} (x) = 
\begin{cases}
	\theta \exp(- \theta x) & \text{ if } x \leq C_{\text{stop}}, \\
	\exp(- \theta C_{\text{stop}}) & \text{ if } x > C_{\text{stop}}.
\end{cases}
\]
For hybrid type-I-censoring we have $C_{\text{stop}} \in \{X_{r:n}, c\}$.

\textbf{Case 1:} $\mathbf{R \geq r}$

If at least $r$ observations occur before time $c$, then $X_{r:n} \leq c$ and therefore
\[
\min \left( X_{r:n}, c \right) = X_{r:n}.
\]
So we stop at the $r$-th observation, before time $c$. 
Then $r$ observations are uncensored and the remaining $n - r$ observations are censored at time $x_{r:n}$. 
In this case, the likelihood function is
\begin{align*}
	L(\theta) 
	&=
	\underbrace{ \prod_{i=1}^r \theta \exp \left( - \theta x_{i:n} \right)}_{\text{uncensored}} \cdot \underbrace{\exp(- \theta x_{r:n})^{n-r}}_{\text{censored}} \\
	&=
	\theta^r \exp \left( - \theta \left(  \sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n} \right) \right).
\end{align*}
Hence the MLE is
\begin{align*}
	\theta^\ast_{\text{hybrid I}} = \frac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}}.
\end{align*}

\textbf{Case 2:} $\mathbf{R < r}$

If fewer than $r$ observations occur before time $c$, then $X_{r:n} > c$ and therefore
\[
\min \left( X_{r:n}, c \right) = c.
\]
So we stop the experiment at time $c$. 
Then $R$ observations are uncensored and the remaining $n - R$ observations are censored at time $c$. 
In this case, the likelihood function is
\begin{align*}
	L(\theta) 
	&=
	\underbrace{ \prod_{i=1}^R \theta \exp \left( - \theta x_{i:n} \right)}_{\text{uncensored}} \cdot \underbrace{\exp(- \theta c)^{n-R}}_{\text{censored}} \\
	&=
	\theta^R \exp \left( - \theta \left( \sum_{i = 1}^R x_{i:n} + (n-R) c \right) \right).
\end{align*}

For $R \geq 1$, the same calculation as in Case 1 gives the MLE
\begin{align*}
	\theta^\ast_{\text{hybrid I}} = \frac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c}.
\end{align*}
For $R = 0$, we have $L(\theta) = \exp(-\theta n c)$. This function is strictly decreasing in $\theta$, so it has no maximum on $(0, \infty)$. 
In this case the MLE does not exist.

All in all, for hybrid type I censoring the MLE is given by
\begin{align} \label{MLE hybrid I}
	\theta^\ast_{\text{hybrid I}} = 
	\begin{cases}
		\dfrac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c} & \text{for } R \in \{1, \ldots , r-1\}, \\[3ex]
		\dfrac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}} & \text{for } R \in \{r, \ldots , n\},
	\end{cases}
\end{align}
and it does not exist for $R = 0$. The problem that the MLE does not exist in the case $R = 0$ is solved by hybrid type II censoring.

\subsection{Example: Hybrid Type-II-censored Exponentially Distributed Lifetimes}

For a hybrid type-II censoring scheme, we set 
\[
C_{\text{stop}} = \max \left( X_{r:n}, c \right),
\]
which means, that in the case of $R = 0$, we still need to continue observing $r$ observations. So the MLE exists in every case. 

\textbf{Case 1:} $\mathbf{R \geq r}$ \\

As seen in the previous section, in this case at least $r$ observations occur before time $c$, so $X_{r:n} \leq c$.
Hence
\[
C_{\text{stop}} = \max \left( X_{r:n}, c \right) = c.
\]
So we stop at time $c$. 
Then $R$ observations are uncensored and the remaining $n - R$ observations are censored at time $c$.

Analogously as before, we can conclude that the MLE is given by 
\begin{align*}
	\theta^\ast_{\text{hybrid II}} = \frac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c}.
\end{align*}

\textbf{Case 2:} $\mathbf{R < r}$ \\
In this case it holds $X_{r:n} > c$, meaning
\[
C_{\text{stop}} = \max \left( X_{r:n}, c \right) = X_{r:n}.
\]
resulting in the MLE 
\begin{align*}
	\theta^\ast_{\text{hybrid II}} = \frac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}}.
\end{align*}

All in all, for hybrid type-II censoring the MLE is given by
\begin{align} \label{MLE hybrid II}
	\theta^\ast_{\text{hybrid II}} = 
	\begin{cases}
		\dfrac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}} & \text{for } R \in \{0, \ldots , r-1\}, \\[3ex]
		\dfrac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c} & \text{for } R \in \{r, \ldots , n\}.
	\end{cases}
\end{align}

\section{Confidence Intervals for the Binomial Distribution: Binomial Inversion}
\textcolor{red}{($\star$) was mit diesem Section machen? passt nicht ganz in die DGF Section}

Let $N \sim \text{Bin}(n, p)$ with known $n \in \mathbb{N}$ and unknown $p \in [0, 1]$.
In this section, we present two \textcolor{red}{($\star$)} methods for constructing a CI for the parameter $p$.
Throughout, let $\alpha \in (0,1)$ be the significance level and let $k$ denote the observed value of $N$.


The idea of binomial inversion is to find all parameter values $p$ for which the observed value $k$ is not too extreme.
A value $p$ is rejected if the observed $k$ lies in one of the tails of the $\text{Bin}(n,p)$ distribution with probability at most $\alpha/2$.
The CI consists of all values of $p$ that are not rejected.
This is called ``inverting'' the binomial distribution function, since we solve for the parameter $p$ and not for the data.

For fixed $k$, the two tail probabilities
\[
\mathbb{P}_p(N \geq k) = \sum_{j=k}^{n} \binom{n}{j} p^j (1-p)^{n-j}
\qquad \text{and} \qquad
\mathbb{P}_p(N \leq k) = \sum_{j=0}^{k} \binom{n}{j} p^j (1-p)^{n-j}
\]
are continuous in $p$. 

For an observed value $N = k$, define the lower bound $p_L(k)$ and the upper bound $p_U(k)$ as the solutions of
\begin{align*}
	\mathbb{P}_{p_L}(N \geq k) = \sum_{j=k}^{n} \binom{n}{j} p_L^{\,j} (1-p_L)^{n-j} &= \frac{\alpha}{2}, \\
	\mathbb{P}_{p_U}(N \leq k) = \sum_{j=0}^{k} \binom{n}{j} p_U^{\,j} (1-p_U)^{n-j} &= \frac{\alpha}{2},
\end{align*}
with the conventions $p_L(0) = 0$ and $p_U(n) = 1$.
The $(1-\alpha)$ confidence interval is
\[
\text{CI}_{1-\alpha}(k) = \big[\, p_L(k),\; p_U(k) \,\big].
\]
This interval is also called the \emph{Clopper--Pearson interval}.


The CI is exact in the sense that its coverage is at least $1-\alpha$ for all $p$ and all $n$.
However, it is \emph{conservative}: since $N$ is discrete, the tail probabilities cannot be exactly equal to $\alpha/2$ for the observed data, and the true coverage is usually strictly larger than $1-\alpha$.
As a consequence, the interval is often wider than necessary.

\begin{example}
	Let $n = 20$ and $p$ be very high, such that we get the realisation $k = 20$. 
	\[
		\mathbb{P}_{p_L} \left( N \geq 20 \right) 
		=
		\sum_{j = 20}^{20} \binom{20}{j} p_L^j (1- p_L)^{20-j} 
		=
		p_L^{20} 
		\overset{!}{=}
		\frac{\alpha}{2}
	\]
	so we can express $p_L = \left( \frac{\alpha}{2} \right)^{\frac{1}{20}}$ and $p_U = 1$ so 
	\[
		\text{CI}_{1- \alpha} (20) = \left[ \left( \frac{\alpha}{2} \right)^{\frac{1}{20}}, 1 \right]
	\]
	for example with $\alpha = 0.05$ is the CI given by $[0.83, 1]$ which is a good estimate for high $p$.
	
\end{example}
