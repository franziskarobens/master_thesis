If we have a random variable $T$ with parameter $\theta$, a data-generating function (DGF) with respect to $T$ is a function $\tau$,
$\tau(\mathbf{U}, \theta)$,
such that $T \overset{d}{=} \tau(\mathbf{U}, \theta)$, where $\mathbf{U}$ is a random vector independent of the parameter $\theta$. Intuitively, one can express this as
\[
\text{data} = \text{randomness} + \text{parameter},
\]
where the parameter is separated from the randomness. The random vector $\mathbf{U}$ must not depend on $\theta$. Thus, the randomness and the parameter $\theta$ are separated.

\[
\mathbf{U} \overset{\tau(\cdot, \theta)}{\longrightarrow} T.
\]

\textcolor{red}{($\star$) 
	Example:
	\begin{enumerate}
		\item Normal distribution
		\item wo es keine DGF gibt
	\end{enumerate}
}

\begin{example}[Example: DGF, Normal distribution] \label{example: DGF normal} 
	Let $T \sim \mathcal{N}(\theta, 1)$. Then we can say that 
	\[
	T \sim \theta + U
	\]
	where $U \sim \mathcal{N}(0, 1)$. So $U$ does not depend on the parameter $\theta$. We can define the DGF
	\begin{equation*}
		\tau(U, \theta) = \theta + U.
	\end{equation*}
	
	If $\theta$ is given, the DGF $\tau$ generates simulated data of $T$. Assume we observe $T= t$ and $U = u$ is given. Then we can get a best estimate of the parameter $\theta$ by inverting $\tau$:
	\begin{align*}
		t &= \tau(u, \theta) = \theta + u \\
		\implies \theta &= t - u
	\end{align*}
	So the inversion is $\hat \theta (u, t) = t-u$. 
	
	In this example, the inversion works perfectly, because $\tau$ is as a function of $\theta$ bijective and therefore invertible. There are also DGFs which can not be inverted, as one can see in Example \ref{example: non invertible DGF}.
	
\end{example}

\begin{example}[Example: DGF with $n > 1$] \label{example:dgf with n large}
 	We can generalize Example \ref{example: DGF normal} by defining
	$T = \sum_{i=1}^n X_i$ where $X_1, \ldots , X_n \overset{\text{iid}}{\sim} \mathcal{N}(\theta, 1)$.  Again use the interpretation as in Example \ref{example: DGF normal}: $X_i = \theta + U_i$ for $i \in \{1, \ldots, n\}$ with $U_1, \ldots , U_n \overset{\text{iid}}{\sim} \mathcal{N}(0, 1)$. Then it holds $T = \sum_{i=1} ^n \theta + U_i = n \theta + \sum_{i=1} ^n U_i$, so the DGF is defined as 
	\begin{align} \label{dgf of multi normal}
	    \tau (\mathbf{U}, \theta )  = n \theta + \sum_{i=1} ^n U_i.
	\end{align}
	This again is invertible with
	\begin{align} \label{dgf of multi normal fiducial distn}
	    \hat \theta(\mathbf{u}, t ) = \frac{t - \sum_{i=1} ^n u_i}{n}.
	\end{align}

Since $\tau$ is strictly monotonically increasing in $\theta$, we can easily solve the equation $\tau(\mathbf{U}, \theta) = t$ for $\theta$, obtaining $\hat{\theta}(\mathbf{U}, t)$, as illustrated in Figure~\ref{fig:dgf example inverse}.

\begin{example}[Non invertible DGF] \label{example: non invertible DGF}
	
	Let \(T \sim \text{Bin}(n,\theta)\) with success probability $\theta \in \Theta = [0, 1]$.
	
	Let \(\mathbf U=(U_1,\ldots,U_n)\) be a random vector whose components are independent and uniformly distributed on \([0,1]\). For each \(i \in \{1, \ldots , n\}\),  it holds
	$U_i \leq \theta$ if and only if
	$\mathds{1}(U_i\leq\theta)=1$
	and therefore since \(U_i\) is uniformly distributed on \([0,1]\),
	\[
	\mathbb P\left(\mathds{1}(U_i \leq \theta)=1\right)
	=
	\mathbb P(U_i \leq \theta)
	=
	\theta.
	\]
	Thus,
	\(
	\mathds{1}(U_i \leq \theta)\sim\operatorname{Bin}(1,\theta)
	\)
	for all \(i\in\{1,\ldots,n\}\). Since the random variables $U_1, \ldots , U_n$ are independent and therefore also their transformations $\mathds{1}(U_1 \leq \theta), \ldots ,  \mathds{1}(U_n \leq \theta)$, their sum is binomially distributed too. Hence,
	\[
	\sum_{i=1}^n \underbrace{\mathds{1}(U_i \leq \theta)}_{\sim \text{Bin}(1, \theta)}
	\sim \operatorname{Bin}(n,\theta).
	\]
	
	Using this result, we define the DGF by
	\begin{align} \label{dgf in clopper pearson case}
		\tau(\mathbf U,\theta)
		=
		\sum_{i=1}^n \mathds{1}(U_i\leq\theta)
	\end{align}
	and by construction it holds
	\[
	T\overset{d}{=}\tau(\mathbf U,\theta).
	\]
	
	To illustrate the DGF, let us fix \(\mathbf U\) and consider \(\tau(\mathbf U,\theta)\) as a function of \(\theta\).
	
	For example, let \(n=2\), and let \(U_{1:2}\) and \(U_{2:2}\) denote the order statistics of \(\mathbf U\), that is,
	\(
	U_{1:2}=\min(U_1,U_2)
	\)
	and
	\(
	U_{2:2}=\max(U_1,U_2).
	\)
	Then the DGF is
	\begin{align}
		\tau (\mathbf U, \theta)
		&=
		\sum_{i = 1}^2 \mathds{1} (U_i \leq \theta)
		=
		\mathds{1} (U_1 \leq \theta) +  \mathds{1} (U_2 \leq \theta)
		\notag \\ 
		&= 
		\mathds{1} (U_{1:2} \leq \theta) +  \mathds{1} (U_{2:2} \leq \theta) 
		=
		\begin{cases}
			0, \text{ if } \theta \in [0, U_{1:2}] \\
			1, \text{ if } \theta \in (U_{1:2}, U_{2:2}] \\
			2, \text{ if } \theta \in (U_{2:2}, 1]
		\end{cases}
		\label{dgf for clopper pearson example}
	\end{align}
	
	Figure~\ref{fig:clopper_pearson_interval_dgf} shows this function schematically.
	\begin{figure}[htbp]
		\centering
		\includegraphics[width=\textwidth]{figures/lindqvist/clopper_pearson_interval_dgf.pdf}
		\caption{DGF for the example given in \eqref{dgf for clopper pearson example}.}
		\label{fig:clopper_pearson_interval_dgf}
	\end{figure}
	

	For example if we observe $t = 1$, then $\hat \theta (\mathbf U, 1) \in (U_{1:2}, U_{2:2}]$. For any observation $t \in \{0, 1, 2\}$, we have infinitely many solutions for the parameter $\hat \theta (\mathbf U, t)$. In particular, the DGF $\tau$ is not invertible.
	
\end{example}


Lindqvist~\cite{Lindqvist2025simulated} presents Theorem~1 in his paper, which provides a method for constructing an upper confidence bound for the parameter $\theta$ based on a DGF.
\begin{theorem}\label{theorem:dgf-upper bound}
	Let $T$ be a random variable with parameter $\theta \in \Theta$ and let $\tau$ be the corresponding DGF.
	Define \[\bar \theta ( \mathbf u, t) = \sup \{ \theta: \tau (\mathbf u, \theta) \leq t \}\] and let $a(t)$ be a left continuous function such that
	\begin{align} \label{precondition theorem 1}
		\mathbb{P} \left( \bar \theta (\mathbf U, t)  \geq a(t)\right) \leq \alpha
	\end{align}
	for all $t$ and $\alpha \in (0, 1)$. 
	Then $a(T)$ is an upper confidence bound for $\theta$ that is $\mathbb{P} \left( \theta \geq a(T) \right) \leq \alpha $ for all $\alpha \in (0, 1)$.
\end{theorem}


For fixed observations \(T=t\) and \(\mathbf U=\mathbf u\), \(\bar{\theta}(\mathbf u,t)\) represents the supremum of all parameter values \(\theta\) for which \(\tau(\mathbf u,\theta)\leq t\). This construction does not require the DGF \(\tau\) to be monotone or continuous, allowing the method to accommodate functions with jumps. To illustrate the construction of $a$, we consider the following example.

\begin{example}
	Suppose $T \sim \text{Exp} \left( \frac{1}{\theta} \right)$ and $U \sim \text{Exp}(1)$. Then we can express $T$ in terms of the function $\tau$ if we define it as 
	\[ \tau (U , \theta ) = \theta \cdot U\]
	because then 
	\[
		\mathbb{P} (\tau (U , \theta ) \leq x) = 
		\mathbb{P} \left( \theta \cdot U \leq x \right) =
		\mathbb{P} \left( U \leq \frac{x}{\theta} \right) = 
		1 - \exp \left( - \frac{x}{\theta} \right)
	\]	
	which is the CDF of an exponentially distributed random variable with parameter $ \frac{1}{\theta}$. So it holds $T \overset{d}{=} \tau (U, \theta)$. Now we calculate $\bar \theta (u, t)$ as
	\[
		\bar \theta (u, t) = 
		\sup \left\{ \theta : \tau (u, \theta) \leq t \right\}
		=
		\sup \left\{ \theta : \theta \cdot u \leq t \right\}
		=
		\sup \left\{ \theta : \theta \leq \frac{t}{u} \right\}
		=
		\frac{t}{u}.
	\]
	Let $\alpha \in (0, 1)$ be arbitrary.
	Now the next objective is to find the function $a$ such that $\mathbb{P} \left( \bar \theta (U, t) \geq a(t) \right) = \alpha$.
	\begin{align*}
		&\mathbb{P} \left( \bar \theta (U, t) \geq a(t) \right) = \alpha
		\\
		\Leftrightarrow \quad &
		\mathbb{P} \left( \frac{t}{U} \geq a(t) \right) = \alpha 
		\\
		\Leftrightarrow \quad &
		\mathbb{P} \left( U \leq \frac{t}{a(t)} \right) = \alpha 
		\\
		\Leftrightarrow \quad &
		1 - \exp \left( - \frac{t}{a(t)} \right) = \alpha 
		\\
		\Leftrightarrow \quad &
		a(t) = \frac{- t}{\ln(1- \alpha)}
	\end{align*}
	The function $a$ is left continuous because it is a linear function of \(t\), so therefore we can use Theorem~\ref{theorem:dgf-upper bound} on the function $a$.
	So after observing $T$, one can say that 
	\[
		\left( 0, a(T) \right] = \left( 0,  \frac{-T}{\ln(1- \alpha)} \right]
	\]
	is a one-sided $(1-\alpha)$-confidence interval for $\theta$. For example for $\alpha = 0.1$ the upper $(1-\alpha)$-confidence bound is given by $\frac{-T}{\ln(1 - 0.1)} \approx 9.49 T$.
\end{example}

\begin{proof}
	Let $\alpha \in (0, 1)$ be arbitrary but fixed. Let $\bar \theta$ be defined as in Theorem~\ref{theorem:dgf-upper bound}, and let $a$ be a left-continuous function such that
	\begin{align*}
		\mathbb{P} \left( \bar \theta (\mathbf U, t) \geq a(t) \right) \leq \alpha
	\end{align*}
	for all $t$.
	We need to show that $a(T)$ is an upper confidence bound, i.\,e.\
	\begin{align} \label{to show theorem 1}
		\mathbb{P} \left( \theta \geq a(T) \right) \leq \alpha
	\end{align}
	for all $\theta \in \Theta$.
	
	Let us define
	\[
	a^{-1} (\theta ) = \sup \{ t : a(t) \leq \theta \}.
	\]
	
	First, we show five statements in step i) to step v). Then, we use these statements to prove the inequality \eqref{to show theorem 1}.
	
	\textbf{Step (i):} \\	
	Assume $\theta \geq a(T)$. Then $T \in \{ t: a(t) \leq \theta \}$. Since $T$ is an element of this set, $T$ has to be smaller or equal to the supremum of this set, i.e. 
	\[
		T \leq \sup \{ t: a(t) \leq \theta \} = a^{-1}(\theta).
	\]
	This is a pure consequence of the definition of the supremum of a set. All in all, the inequality $\theta \geq a(T)$, implies $T \leq a^{-1}(T)$. 
	So we can conclude that
	\[
		\{ \theta \geq a(T) \} \subseteq \{ T \leq a^{-1}(\theta) \}.
	\]
	and therefore 
	\begin{align} \label{theorem 1: step i}
		\mathbb{P} \left( \theta \geq a(T) \right) \leq 
		\mathbb{P} \left( T \leq a^{-1} (\theta) \right).
	\end{align}	
	
	
	\textbf{Step (ii):} \\
	
	Since $T \overset{d}{=} \tau (\mathbf U, \theta)$ because $\tau$ is a DGF of $T$, we can say that the equality 
	\begin{align} \label{theorem 1: step ii}
		\mathbb{P} \left( (T \leq a^{-1}(\theta)) \right)
		=
		\mathbb{P} \left( (\tau (\mathbf U, \theta) \leq a^{-1}(\theta)) \right)
	\end{align}
	holds.
	
	\textbf{Step (iii):} \\
	Assume $ \tau ( \mathbf U, \theta ) \leq a^{-1}(\theta)$. Define $t' := a^{-1}(\theta)$, then it holds $ \tau ( \mathbf U, \theta ) \leq t'$. So, we can observe that $\theta \in \{\theta' : \tau(\mathbf U, \theta') \leq t'\}$. Again, with the properties of the supremum, we get
	\[
		\theta \leq \sup \{\theta' : \tau(\mathbf U, \theta') \leq t'\}
		= \bar \theta (\mathbf U, t').
	\]
	All in all, it holds that $\tau (\mathbf U, \theta) \leq t'$ implies that $\theta \leq \bar \theta (\mathbf U, t')$. We can use this to know that 
	\[
			\left\{ 
				\tau (\mathbf U, \theta) \leq a^{-1} (\theta) 
			\right\}
		\subseteq
			\left\{ 
				\bar \theta \left( \mathbf U, a^{-1}(\theta) \right)  \geq \theta
			\right\}.
	\]
	and obtain the inequality
	\begin{align} \label{theorem 1: step iii}
		\mathbb{P} \left( \tau (\mathbf U, \theta) \leq a^{-1} (\theta) \right)
		\leq 
		\mathbb{P} \left( \bar \theta \left( \mathbf U, a^{-1}(\theta) \right)  \geq \theta \right).
	\end{align}
	

	
	
	\textbf{Step (iv):} \\
	We want to show that $a\big(a^{-1}(\theta)\big) \le \theta$.
	This is where left-continuity of $a$ is used. Let
	\[
	t^\ast := a^{-1}(\theta) = \sup\{t : a(t) \le \theta\}.
	\]
	If the defining set $\{t : a(t) \le \theta\}$ has a maximum, i.e. contains $t^\ast$ itself, then trivially $a(t^\ast) \le \theta$ and we are done.
	
	Otherwise, $t^\ast$ is only a limit point of the set. There is a sequence $t_n \nearrow t^\ast$ with $a(t_n) \le \theta$ for every $n$. Left-continuity of $a$ means
	\[
	a(t_n) \longrightarrow a(t^\ast) \quad
	\]
	as $t_n \nearrow t^\ast$.
	Since $a(t_n) \le \theta$ for all $n$, and the limit of a sequence bounded above by $\theta$ is itself smaller or equal to$\theta$ and we obtain
	\(
	a(t^\ast) \le \theta,
	\)
	so
	\(
	a\big(a^{-1}(\theta)\big) \le \theta.
	\)
	So we can conclude that 
	\begin{align} \label{theorem 1: step iv}
		\mathbb{P} \left( \bar \theta (\mathbf U, a^{-1}(\theta)) \geq \theta \right)
		\geq 
		\mathbb{P} \left( \bar \theta (\mathbf U, a^{-1}(\theta)) \geq a \left( a^{-1} (\theta) \right) \right).
	\end{align}
	
	
	\textbf{Step (v):} \\
	As a last step, we show a direct instance of the hypothesis~\eqref{precondition theorem 1}
	applied with the particular choice $t = a^{-1}(\theta)$.
	We get 
	\begin{align} \label{theorem 1: step v}
		\mathbb{P}_U\Big( \theta\big(U, a^{-1}(\theta)\big) \ge a\big(a^{-1}(\theta)\big) \Big) \le \alpha.
	\end{align}
	
	\textbf{Conclusion:} \\
	In the end, we can combine the knowledge derived in step (i) to step (v). It holds
	\begin{align*}
		\mathbb{P} \left( \theta \geq a(T) \right)
		&\overset{\eqref{theorem 1: step i}}{\leq}
		\mathbb{P} \left( T \leq a^{-1}(\theta) \right) \\
		&\overset{\eqref{theorem 1: step ii}}{=}
		\mathbb{P} \left( \tau(\mathbf U, \theta) \leq a^{-1}(\theta) \right) \\
		&\overset{\eqref{theorem 1: step iii}}{\leq}
		\mathbb{P} \left[
		\bar \theta (\mathbf U, a^{-1}(\theta)) \geq \theta
		\right] \\
		&\overset{\eqref{theorem 1: step iv}}{\leq}
		\mathbb{P} \left[ 
		\bar \theta (\mathbf U, a^{-1}(\theta)) \geq a^{-1}(a(\theta))
		\right] \\
		&\overset{\eqref{theorem 1: step v}}{\leq}
		\alpha
	\end{align*}
	and this shows that $a(T)$ is an upper confidence bound with confidence level $1 - \alpha$.
	
\end{proof}

\begin{corollary} \label{corollary:dgf-lower bound}
	Let $T$ be a random variable with parameter $\theta \in \Theta$ and let $\tau$ be the corresponding DGF. Define
	\[
		\underline \theta (\mathbf u, t) = \inf \{\theta: \tau(\mathbf u, \theta) \geq t\}
	\]
	and let $b(t)$ be a right continuous function such that
	\[
	\mathbb{P} \left( \underline \theta (\mathbf U, t)  \leq b(t)\right) \leq \alpha
	\]
	for all $t$ and $\alpha \in (0, 1)$. 
	Then $b(T)$ is a lower confidence bound for $\theta$ that is $\mathbb{P} \left( \theta \leq b(T) \right) \leq \alpha $ for all $\alpha \in (0, 1)$.
\end{corollary}

Let \(T \sim \operatorname{Bin}(n,\theta)\), and let \(t \in \{0,\ldots,n\}\) be the observed value of \(T\). The goal is to construct a confidence interval for the success probability \(\theta \in \Theta=[0,1]\).

We already analysed this situation in Example~\ref{example: non invertible DGF}, so we know that the function given in \eqref{dgf in clopper pearson case} is a valid DGF of $T$. 

To use Theorem~\ref{theorem:dgf-upper bound}, we define
\[
\bar \theta (\mathbf u, t) 
=
\sup \{ \theta : \tau(\mathbf u, \theta) \leq t \}
=
\sup \left\{ \theta : \sum_{i = 1}^n \mathds{1}(u_{i:n} < \theta) \leq t \right\}
=
\begin{cases}
	u_{1:n} &\text{ for } t = 0, \\
	u_{2:n} &\text{ for } t = 1, \\
	\vdots & \\
	u_{(n-1):n} &\text{ for } t = n-2, \\
	u_{n:n} &\text{ for } t = n-1, \\
	1 &\text{ for } t = n, \\
\end{cases}
\]
All in all, $\bar \theta (\mathbf u, t) = u_{(t+1):n}$ for all $t \in \{1, \ldots , n\}$ whereas we define $u_{(n+1):n} := 1$.



Next, we want to find a left continuous function $a$ such that
\begin{align} \label{condition for theorem 1}
	\mathbb{P} \left( \bar \theta (\mathbf U, t) \geq a(t) \right) \leq \alpha
\end{align}
for all $t\in \{0, \ldots , n\}$ and $\alpha \in (0, 1)$.

To construct $a(t)$, we consider all possible values of $t$ and choose $a(t)$ accordingly. Assume that $t\in\{0,\ldots,n-1\}$. We require that
\begin{align}
	\alpha
	\overset{!}{=}&
	\mathbb{P}\left(
	\bar{\theta}(\mathbf{U},t) \leq a(t)
	\right) \\
	=&
	\mathbb{P}\left(
	U_{(t+1):n} \leq a(t)
	\right).
	\label{criteria clopper pearson}
\end{align}
Since the order statistics of independent standard uniformly distributed random variables follow Beta distributions, i.e.,
\[
U_{k:n}\sim\operatorname{Beta}(k,n-k+1),
\]
for all $k \in \{1, \ldots, n\}$,
we can use the quantiles of the Beta distribution to determine $a(t)$.
Let $\text{Beta}_\beta (p, q)$ denote the $\beta$ quantile of the $\text{Beta}(p, q)$ distribution. 
If we define
\[
a(t)=\text{Beta}_{1-\alpha}(t+1,n-t),
\]
this choice of $a(t)$ satisfies the criterion in~\eqref{criteria clopper pearson}.

On the other hand, if we want to construct a lower $(1-\alpha)$ confidence interval based on Corollary~\ref{corollary:dgf-lower bound}.
First, we observe that 

\begin{align} \label{precondition for corollary 1}
	\underline{\theta}(\mathbf U, t)
	=
	\inf \{ \theta \in \Theta : \tau (\mathbf U , \theta ) \geq t \}
	=
	U_{(t:n)}.
\end{align}


So if we want to find a right continuous function $b$, with 
\begin{align*}
	\mathbb{P} \left( \underline{\theta} (\mathbf U, t) \leq b(t) \right) \leq \alpha 
\end{align*}
for all $t\in \{0, \ldots , n\}$ and $\alpha \in (0, 1)$, we can use that $\underline{\theta}$ is given by \eqref{precondition for corollary 1} and try to find 
\begin{align*}
	&\mathbb{P} \left( \underline{\theta} (\mathbf U, t) \leq b(t) \right) 
	=
	\mathbb{P} \left( U_{(t:n)} \leq b(t) \right) \overset{!}{=}& \alpha \\
	\Longleftrightarrow&
	b(t) =& B(\alpha; t, n+1-t).
\end{align*}

The function $b$ is right continuous because \textcolor{red}{($\star$)}. 
With Corollary~\ref{corollary:dgf-lower bound}, we can conclude that for a given observation $T = t$, the interval $[b(T), 1] = \left[ B(\alpha; T, n+1-T), 1 \right]$ is a lower $(1- \alpha)$ confidence interval for $\theta$.




\section{Example from Lindqvist Section 5: DGF of Type I Censored Exponentially Distributed Lifetimes}

\textcolor{red}{($\star$) das sollte kein ganzer abschnitt sein, oder? Sondern nur example, oder andere examples auch zu einer section machen}

Let $X_1 , \ldots , X_n$ be independent exponentially distributed random variables with unknown hazard rate $\theta>0$. Let $Y_1, \ldots , Y_n$ be the zugehörigen type I censored data with fixed censoring time $c>0$, i. e. $Y_i = \min(X_i, c)$ for all $i \in \{1, \ldots , n\}$. Define the number of uncensored obervations as
\[
	R = \sum_{i = 1}^n \mathds{1} \left( Y_i = X_i \right) = \sum_{i = 1}^n \mathds{1} _{(0, c]} (X_i).
\] 
The MLE of $\theta$ is of interest and we can define $T$ such that $T$ matches the MLE.
The density of $Y_i$ for $i \in \{1 , \ldots , n\} $ is given by
\[
	f_{Y_i}(y) = 
	\begin{cases}
		\theta \exp(- \theta y),  & \text{ if } y \leq c \\
		\exp(- \theta c),  & \text{ if } y > c
	\end{cases}
\]
since if $Y_i$ is uncensored $\mathbb{P} \left( Y_i = y \right) = \mathbb{P} \left( X_i = y \right) = f_{X_i}(y)$ and if $Y_i$ is censored
$\mathbb{P} \left( Y_i = y \right) = \mathbb{P} \left( X_i > c \right) = 1 - F_{X_i}(c) = \exp(- \theta c)$.
The likelihood function of the censored sample is given by
\begin{align}
	L(\theta) = \prod_{i=1}^n f_{Y_i}(y_i)
	&=
	\underbrace{\prod_{i=1}^R \theta \exp(- \theta y_{i:n}) }_{\text{uncensored}} 
	\cdot 
	\underbrace{\prod_{i=R+1}^n \exp(- \theta c) }_{\text{censored}}  \\
	&=
	\theta^R \exp \left( - \theta \left[ \underbrace{\sum_{i = 1}^R y_{i:n} + (n - R) c}_{= \sum_{i = 1}^n y_i} \right] \right)
\end{align}
For simplicity, lets define 
\[
	S = \sum_{i = 1}^n y_i
\]
Then log likelihood is given by 
\begin{align*}
	l(\theta) = \ln L(\theta) = R \log(\theta) - \theta S
\end{align*}
and we can determine the MLE if we set the derivate of $l$ equal to zero
\begin{align}
	l'(\theta) = 0 \Longleftrightarrow \frac{R}{\theta} - S = 0 \Longleftrightarrow \theta = \frac{R}{S}.
\end{align}
We can easily define 
\begin{align} \label{T for exp distn}
	T = \frac{R}{S} = \frac{
	\sum_{i = 1}^n \mathds{1}(X_i \leq c)
	}{
	\sum_{i = 1}^n \min(X_i, c)}
\end{align}
To seperate the randomness and the parameter $\theta$ which is now not seperated since $X_i$ is a random variable dependent on the parameter $\theta$, we can use a trick. Define $U_1, \ldots , U_n $ as independent standard exponentially distributed random variables. It is a standard result that $\frac{U_i}{\theta}$ is exponentially distributed with parameter $\theta$. So $X_i \overset{d}{=} \frac{U_i}{\theta}$ and we can substitute this into \eqref{T for exp distn}. 
So if we define 
\begin{align}
	\tau(\mathbf U, \theta)
	= 
	\frac{
	\sum_{i = 1}^n \mathds{1}(\frac{U_i}{\theta} \leq c)
	}{
	\sum_{i = 1}^n \min(\frac{U_i}{\theta}, c)}
\end{align}
it holds $T \overset{d}{=} \tau(\mathbf U, \theta)$, so $\tau$ is a DGF for the MLE $T$. 


Since $\frac{U_i}{\theta} < c$ if and only if $\frac{U_i}{c} < \theta$ for all $i \in \{1, \ldots , n\}$, and the ordering of the summands does not matter, we can use the order statistics
\begin{align} 	\label{DGF for MLE of Exp distn}
	\tau(\mathbf U, \theta) = \frac
	{
		\sum_{i = 1}^n  \mathds{1} \left( \frac{U_{i:n}}{c} < \theta \right)
	}{
		\sum_{i = 1}^n \min \left(  \frac{U_{i:n}}{\theta} , c \right)
	}
\end{align}


To illustrate the DGF \eqref{DGF for MLE of Exp distn}, we first consider the simple case $n = 2$. In this case, the DGF takes the form
\begin{align} \label{dgf for n = 2}
	\tau (\mathbf U, \theta)
	=
	\frac
	{
		\mathds{1} \left( \frac{U_{1:2}}{c} < \theta \right)
		+
		\mathds{1} \left( \frac{U_{2:2}}{c} < \theta \right)
	}{
		\min \left(  \frac{U_{1:2}}{\theta} , c \right)
		+
		\min \left(  \frac{U_{2:2}}{\theta} , c \right)
	}. 	
\end{align}



Since the numerator can only take the values $0$, $1$, and $2$, we consider three cases.


\textbf{Case A: $R = 0$}

If $R=0$, the numerator of \eqref{dgf for n = 2} is equal to zero. This means that neither condition $A_1$ nor condition $A_2$ is satisfied, where
\[
A_1 := \left\{\frac{U_{1:2}}{c}<\theta\right\},
\qquad
A_2 := \left\{\frac{U_{2:2}}{c}<\theta\right\}.
\]

Therefore, it holds that 
$\frac{U_{1:2}}{c}\geq\theta$
and
$\frac{U_{2:2}}{c}\geq\theta$.
Since
\(
\frac{U_{1:2}}{c}\leq\frac{U_{2:2}}{c}
\)
and $\theta>0$, these two conditions can be simplified to
\[
\theta\in\left(0,\frac{U_{1:2}}{c}\right].
\]

Since the numerator of \eqref{dgf for n = 2} is zero, the whole fraction is zero. Hence,
\[
\tau(\mathbf U,\theta)=0
\]
for all
$\theta\in\left(0,\frac{U_{1:2}}{c}\right]$.


\textbf{Case B: $R = 1$}

If $R=1$, the numerator of \eqref{dgf for n = 2} is equal to one. Therefore, exactly one of the conditions $A_1$ and $A_2$ is satisfied.

Since
\[
\frac{U_{2:2}}{c}<\theta
\quad\Longrightarrow\quad
\frac{U_{1:2}}{c}<\theta,
\]
because $U_{1:2}\leq U_{2:2}$ by the definition of order statistics, condition $A_2$ implies condition $A_1$. Therefore, if exactly one condition is satisfied, $A_1$ is satisfied but $A_2$ is not. Thus,
\[
\frac{U_{1:2}}{c}<\theta
\qquad\text{and}\qquad
\frac{U_{2:2}}{c}\geq\theta,
\]
which means that
\[
\theta\in
\left(
\frac{U_{1:2}}{c},
\frac{U_{2:2}}{c}
\right].
\]

Hence,
$\mathds{1}(A_1)=1$
and
$\mathds{1}(A_2)=0$.
Therefore, the numerator is equal to $1$.

We can also determine the two minimums in the denominator. Since
\[
\frac{U_{1:2}}{c}<\theta
\leq
\frac{U_{2:2}}{c},
\]
we obtain
\[
c>\frac{U_{1:2}}{\theta}
\geq
c\frac{U_{1:2}}{U_{2:2}}
\]
and
\[
c\frac{U_{2:2}}{U_{1:2}}
>
\frac{U_{2:2}}{\theta}
\geq c.
\]

Therefore,
\[
\min\left(\frac{U_{1:2}}{\theta},c\right)
=
\frac{U_{1:2}}{\theta}
\]
and
\[
\min\left(\frac{U_{2:2}}{\theta},c\right)
=
c.
\]

Thus, for
$\theta\in
\left(
\frac{U_{1:2}}{c},
\frac{U_{2:2}}{c}
\right]$,
the DGF is given by 
\begin{align}
	\tau(\mathbf U,\theta)
	=
	\frac{1}{
		\frac{U_{1:2}}{\theta}+c
	}
	=
	\frac{\theta}{U_{1:2}+c\theta}.
\end{align}


\textbf{Case C: $R = 2$}

If $R=2$, both indicators are equal to one. Therefore, both conditions $A_1$ and $A_2$ are satisfied. In particular,
\[
\frac{U_{2:2}}{c}<\theta,
\]
which implies
\[
\theta\in
\left(
\frac{U_{2:2}}{c},
\infty
\right).
\]

Since
\[
\frac{U_{2:2}}{c}<\theta,
\]
taking reciprocals and multiplying by $U_{1:2}$ and $U_{2:2}$, respectively, gives
\[
c\frac{U_{1:2}}{U_{2:2}}
>
\frac{U_{1:2}}{\theta}
\]
and
\[
c>
\frac{U_{2:2}}{\theta}.
\]

Since
$U_{1:2}\leq U_{2:2}$,
we have
\[
\frac{U_{1:2}}{U_{2:2}}\leq 1.
\]
Therefore,
\[
c
\geq
c\frac{U_{1:2}}{U_{2:2}}
>
\frac{U_{1:2}}{\theta}.
\]

It follows that the minimums in the denominator are
\[
\min\left(\frac{U_{1:2}}{\theta},c\right)
=
\frac{U_{1:2}}{\theta}
\]
and
\[
\min\left(\frac{U_{2:2}}{\theta},c\right)
=
\frac{U_{2:2}}{\theta}.
\]

Thus, for Case C,
\[
\tau(\mathbf U,\theta)
=
\frac{2}{
	\frac{U_{1:2}}{\theta}
	+
	\frac{U_{2:2}}{\theta}
}
=
\frac{2\theta}{U_{1:2}+U_{2:2}}.
\]


Combining all three cases gives
\begin{align} \label{dgf for exp with cases}

=
\left\{
\begin{array}{lll}

	&
	\text{if }
	\theta\in
	\left(
	0,\frac{U_{1:2}}{c}
	\right],
	&
	\qquad (A),
	\\[8pt]
	\displaystyle
	\frac{\theta}{U_{1:2}+c\theta},
	&
	\text{if }
	\theta\in
	\left(
	\frac{U_{1:2}}{c},

$\bar \theta_{\text{full}} (\mathbf U , 2/3)$ and $\hat \theta_{\text{fid}} (\mathbf U , 2/3)$ are random variables itself, so we can determine their distribution.

But first, we need to find the distribution of the order statistics $U_{1:2}$ and $U_{2:2}$. Their underlying random variables are exponentially distributed, i.e. $U_1, U_2 \sim \text{Exp(1)}$.

\begin{align*}
	F_{U_{1:2}}
	&=
	\mathbb{P} \left( U_{1:2} \leq x \right) \\
	&=
	\mathbb{P} \left( \min(U_1, U_2) \leq x \right) \\
	&=
	1 - \mathbb{P} \left( \min(U_1, U_2) > x \right) \\
	&=
	1 - \left( \mathbb{P} \left( U_1 > x \right) \cdot \mathbb{P} \left( U_2 > x \right) \right) \\
	&=
	1 - \left( \left(1 - F_{U_1}(x) \right) \cdot \left(1 - F_{U_2}(x) \right)  \right) \\
	&=
	1 - (\exp(-x) \cdot \exp(-x)) \\
	&=
	1 - \exp(-2x)
\end{align*}
which is the CDF of a $\text{Exp}(2)$ distribution.

The CDF of the second order statistic is given by
\begin{align*}
	F_{U_{2:2}}(x) 
	&=
	\mathbb{P} \left( U_{2:2} \leq x \right) \\
	&=
	\mathbb{P} \left( \max \left( U_1, U_2 \right) \leq x \right) \\
	&=
	\mathbb{P} \left(U_1 \leq x \right) \cdot \mathbb{P} \left(U_1 \leq x \right) \\
	&=
	F_{U_1}(x) \cdot F_{U_2}(x)	\\
	&=
	\left( 1 - \exp(-x) \right)^2.
\end{align*}

Observe that $\bar \theta_{\text{full}} (\mathbf u , 2/3)$ can also be expressed as $\min \left( 2 u_{1:2}, u_{2:2} \right)$. We can use this to determine its distribution 
\begin{align*}
	\mathbb{P} \left( \bar \theta_{\text{full}} (\mathbf U, 2/3) > x \right)
	&=
	\mathbb{P} \left( \min \left( 2 U_{1:2}, U_{2:2} \right) > x \right) \\
	&=
	\mathbb{P} \left( 2 U_{1:2} > x \right) \cdot \mathbb{P} \left( U_{2:2} > x \right) \\
	&=
	\left[ 1 - \mathbb{P} \left( U_{1:2} \leq \frac{x}{2} \right) \right] \cdot 
	\left[ 1 - \mathbb{P} \left( U_{2:2} \leq x \right) \right] \\
	&=
	\left[ 1 - F_{U_{1:2}} \left( \frac{x}{2} \right) \right] \cdot 
	\left[ 1 - F_{U_{2:2}}(x) \right] \\
	&=
	\left[ \exp \left( -2 \frac{x}{2} \right) \right] \cdot \left[ 1 - \left( 1 - \exp(-x) \right)^2 \right] \\
	&= 
	2 \exp(-2x) - \exp(-3x)
\end{align*}

So
\begin{align*}

	=
	1 - 2 \exp(-2x) + \exp(-3x)
\end{align*}
and

\end{align}

Now, we will obtain the distribution of $\hat \theta_{\text{fid}} (\mathbf U, 2/3) = 2 U_{1:2} \; | \; 2 U_{1:2} \leq U_{2:2}$. 


For easier notation, let us define $M = U_{1:2}$.
We can write
\[
U_{2:2} = M + E
\]
with $E \sim \text{Exp}(1)$ because of the memoryless property of the exponential distribution.
The minimum $M = \min(U_1, U_2)$ is the time of the first event. 
At this time, one of the two variables is "used up". 
The other variable has not happened yet. 
By the memoryless property, the remaining waiting time until it happens has the same $\text{Exp}(1)$ distribution as at the start, and it is independent of $M$.
Therefore, the difference $E = U_{2:2} - U_{1:2}$ is $\text{Exp}(1)$-distributed and independent of $M$.

We can also show this rigerously by applying the transformation theorem on the vector $(U_{1:2}m U_{2:2})$ and the function $T(x,y) = (x, y-x)$.
Integrating out the first component gives $U_{2:2} - U_{1:2} = E \sim \text{Exp}(1)$.

Next, note that
\[
2 U_{1:2} \leq U_{2:2}
\iff 2M \leq M + E
\iff M \leq E.
\]
So we can use this notation to describe the fiducial distribution as
\[
\hat{\theta}_{\text{fid}}(\mathbf{U}, 2/3) \sim 2M \;\big|\; M \leq E,
\]
where $M \sim \text{Exp}(2)$ and $E \sim \text{Exp}(1)$ are independent.


\begin{align*}
	\mathbb{P} \left( \hat{\theta}_{\text{fid}}(\mathbf{U}, 2/3) \leq x  \right)
	=
	\mathbb{P} \left( 2M \leq x \;\big|\; M \leq E  \right) \\
	=
	\frac{
		\mathbb{P} \left( 2M \leq x , M \leq E  \right)
	}{
		\mathbb{P} \left( M \leq E  \right)
	}
\end{align*}
The denominator is
\begin{align*}
	\mathbb{P} \left( M \leq E  \right) 
	&=
	\int_0^\infty \mathbb{P} (m \leq E \;\big|\; M = m) \; dM \\
	&=
	\int_0^\infty \mathbb{P} (m \leq E \;\big|\; M = m) f_M(m) \; dm \\
	&=
	\int_0^\infty \exp(-m) \cdot 2 \exp(-2m) \; dm \\
	&=
	\frac{2}{3}
\end{align*}
and for $x \geq 0$, the numerator is
\begin{align*}
	\mathbb{P} \left( 2M \leq x , M \leq E  \right)
	&=
	\mathbb{P} \left( M \leq \tfrac{x}{2} , M \leq E  \right) \\
	&=
	\int_0^{x/2} \mathbb{P} (m \leq E) \, f_M(m) \; dm \\
	&=
	\int_0^{x/2} \exp(-m) \cdot 2 \exp(-2m) \; dm \\
	&=
	\frac{2}{3} \left( 1 - \exp\!\left( -\frac{3x}{2} \right) \right).
\end{align*}
Putting both parts together, we get for $x \geq 0$
\begin{align*}
	\mathbb{P} \left( \hat{\theta}_{\text{fid}}(\mathbf{U}, 2/3) \leq x  \right)
	=
	\frac{\frac{2}{3} \left( 1 - \exp\left( -\frac{3x}{2} \right) \right)}{\frac{2}{3}}
	=
	1 - \exp\!\left( -\frac{3x}{2} \right).
\end{align*}
For $x < 0$ the probability is $0$.
This is the cdf of an exponential distribution with rate $3/2$. Hence
\[
\hat{\theta}_{\text{fid}}(\mathbf{U}, 2/3) \sim \text{Exp}\!\left( \tfrac{3}{2} \right).
\]

In Section 6 of~\cite{Lindqvist2025simulated}, those two methods for retrieving $\theta$ are compared. The simulation study is replicated in Listing~\ref{listing:lindqvist section 6.R} and the theoretical density is added to the plot. The results of Listing~\ref{listing:lindqvist section 6.R} are shown in Figure~\ref{fig:lindqvist Figure 2}. One can see that the theoretical densities approximately match the simulation.

\subsection{Hybrid Type-I censoring}

Assume $X_1, \ldots, X_n \overset{\text{iid}}{\sim} \text{Exp}(\theta)$. We want to find the MLE of the parameter $\theta \in \Theta = (0, \infty)$ based on the censored data $Y_1, \ldots, Y_n$. 
In Section~\ref{...} we analysed this situation for type I and type II censoring. 
Now we apply a hybrid type I censoring scheme to the random variables $X_1, \ldots, X_n$ and derive the MLE.
The experiment stops at the time
\[
C_{\text{stop}} = \min \left( X_{r:n}, c \right),
\]
that is, as soon as either the $r$-th failure has occurred or the time $c$ has been reached, whichever happens first. 
This gives two cases, depending on whether $R \geq r$ or $R < r$.

Recall that each observation contributes to the likelihood as follows
\[
f_{Y_i} (x) = 
\begin{cases}
	\theta \exp(- \theta x) & \text{ if } x \leq C_{\text{stop}}, \\
	\exp(- \theta C_{\text{stop}}) & \text{ if } x > C_{\text{stop}}.
\end{cases}
\]
For type I censoring we have $C_{\text{stop}} = c$. For hybrid type I censoring we have $C_{\text{stop}} \in \{X_{r:n}, c\}$.

\textbf{Case 1:} $\mathbf{R \geq r}$

If at least $r$ observations occur before time $c$, then $X_{r:n} \leq c$ and therefore
\[
\min \left( X_{r:n}, c \right) = X_{r:n}.
\]
So we stop at the $r$-th observation, before time $c$. 
Then $r$ observations are uncensored and the remaining $n - r$ observations are censored at time $x_{r:n}$. 
In this case, the likelihood function is
\begin{align*}
	L(\theta) 
	&=
	\underbrace{ \prod_{i=1}^r \theta \exp \left( - \theta x_{i:n} \right)}_{\text{uncensored}} \cdot \underbrace{\exp(- \theta x_{r:n})^{n-r}}_{\text{censored}} \\
	&=
	\theta^r \exp \left( - \theta \left(  \sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n} \right) \right).
\end{align*}
Hence the MLE is
\begin{align*}
	\theta^\ast_{\text{hybrid I}} = \frac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}}.
\end{align*}

\textbf{Case 2:} $\mathbf{R < r}$

If fewer than $r$ observations occur before time $c$, then $X_{r:n} > c$ and therefore
\[
\min \left( X_{r:n}, c \right) = c.
\]
So we stop the experiment at time $c$. 
Then $R$ observations are uncensored and the remaining $n - R$ observations are censored at time $c$. 
In this case, the likelihood function is
\begin{align*}
	L(\theta) 
	&=
	\underbrace{ \prod_{i=1}^R \theta \exp \left( - \theta x_{i:n} \right)}_{\text{uncensored}} \cdot \underbrace{\exp(- \theta c)^{n-R}}_{\text{censored}} \\
	&=
	\theta^R \exp \left( - \theta \left( \sum_{i = 1}^R x_{i:n} + (n-R) c \right) \right).
\end{align*}

For $R \geq 1$, the same calculation as in Case 1 gives the MLE
\begin{align*}
	\theta^\ast_{\text{hybrid I}} = \frac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c}.
\end{align*}
For $R = 0$, we have $L(\theta) = \exp(-\theta n c)$. This function is strictly decreasing in $\theta$, so it has no maximum on $(0, \infty)$. 
In this case the MLE does not exist.

All in all, for hybrid type I censoring the MLE is given by
\begin{align} \label{MLE hybrid I}
	\theta^\ast_{\text{hybrid I}} = 
	\begin{cases}
		\dfrac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c} & \text{for } R \in \{1, \ldots , r-1\}, \\[3ex]
		\dfrac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}} & \text{for } R \in \{r, \ldots , n\},
	\end{cases}
\end{align}
and it does not exist for $R = 0$. The problem that the MLE does not exist in the case $R = 0$ is solved by hybrid type II censoring.

\subsection{Hybrid Type-II Censoring}

For a hybrid type-II censoring scheme, we set 
\[
C_{\text{stop}} = \max \left( X_{r:n}, c \right),
\]
which means, that in the case of $R = 0$, we still need to continue observing $r$ observations. So the MLE exists in every case. 

\textbf{Case 1:} $\mathbf{R \geq r}$ \\

As seen in the previous section, in this case at least $r$ observations occur before time $c$, so $X_{r:n} \leq c$.
Hence
\[
C_{\text{stop}} = \max \left( X_{r:n}, c \right) = c.
\]
So we stop at time $c$. 
Then $R$ observations are uncensored and the remaining $n - R$ observations are censored at time $c$.

Analogously as before, we can conclude that the MLE is given by 
\begin{align*}
	\theta^\ast_{\text{hybrid II}} = \frac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c}.
\end{align*}

\textbf{Case 2:} $\mathbf{R < r}$ \\
In this case it holds $X_{r:n} > c$, meaning
\[
C_{\text{stop}} = \max \left( X_{r:n}, c \right) = X_{r:n}.
\]
resulting in the MLE 
\begin{align*}
	\theta^\ast_{\text{hybrid II}} = \frac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}}.
\end{align*}

All in all, for hybrid type-II censoring the MLE is given by
\begin{align} \label{MLE hybrid II}
	\theta^\ast_{\text{hybrid II}} = 
	\begin{cases}
		\dfrac{r}{\sum_{i = 1}^r x_{i:n} + (n-r) x_{r:n}} & \text{for } R \in \{0, \ldots , r-1\}, \\[3ex]
		\dfrac{R}{\sum_{i = 1}^R x_{i:n} + (n-R) c} & \text{for } R \in \{r, \ldots , n\}.
	\end{cases}
\end{align}

\section{Confidence Intervals for the Binomial Distribution}

Let $N \sim \text{Bin}(n, p)$ with known $n \in \mathbb{N}$ and unknown $p \in [0, 1]$.
In this section, we present two \textcolor{red}{($\star$)} methods for constructing a CI for the parameter $p$.
Throughout, let $\alpha \in (0,1)$ be the significance level and let $k$ denote the observed value of $N$.

\subsection{Binomial Inversion}

The idea of binomial inversion is to find all parameter values $p$ for which the observed value $k$ is not too extreme.
A value $p$ is rejected if the observed $k$ lies in one of the tails of the $\text{Bin}(n,p)$ distribution with probability at most $\alpha/2$.
The CI consists of all values of $p$ that are not rejected.
This is called ``inverting'' the binomial distribution function, since we solve for the parameter $p$ and not for the data.

For fixed $k$, the two tail probabilities
\[
\mathbb{P}_p(N \geq k) = \sum_{j=k}^{n} \binom{n}{j} p^j (1-p)^{n-j}
\qquad \text{and} \qquad
\mathbb{P}_p(N \leq k) = \sum_{j=0}^{k} \binom{n}{j} p^j (1-p)^{n-j}
\]
are continuous in $p$. 

For an observed value $N = k$, define the lower bound $p_L(k)$ and the upper bound $p_U(k)$ as the solutions of
\begin{align*}
	\mathbb{P}_{p_L}(N \geq k) = \sum_{j=k}^{n} \binom{n}{j} p_L^{\,j} (1-p_L)^{n-j} &= \frac{\alpha}{2}, \\
	\mathbb{P}_{p_U}(N \leq k) = \sum_{j=0}^{k} \binom{n}{j} p_U^{\,j} (1-p_U)^{n-j} &= \frac{\alpha}{2},
\end{align*}
with the conventions $p_L(0) = 0$ and $p_U(n) = 1$.
The $(1-\alpha)$ confidence interval is
\[
\text{CI}_{1-\alpha}(k) = \big[\, p_L(k),\; p_U(k) \,\big].
\]
This interval is also called the \emph{Clopper--Pearson interval}.


The CI is exact in the sense that its coverage is at least $1-\alpha$ for all $p$ and all $n$.
However, it is \emph{conservative}: since $N$ is discrete, the tail probabilities cannot be exactly equal to $\alpha/2$ for the observed data, and the true coverage is usually strictly larger than $1-\alpha$.
As a consequence, the interval is often wider than necessary.

\begin{example}
	Let $n = 20$ and $p$ be very high, meaning we get the realisation $k = 20$. 
	\[
		\mathbb{P}_{p_L} \left( N \geq 20 \right) 
		=
		\sum_{j = 20}^{20} \binom{20}{j} p_L^j (1- p_L)^{20-j} 
		=
		p_L^{20} 
		\overset{!}{=}
		\frac{\alpha}{2}
	\]
	so we can express $p_L = \left( \frac{\alpha}{2} \right)^{\frac{1}{20}}$ and $p_U = 1$ so 
	\[
		\text{CI}_{1- \alpha} (20) = \left[ \left( \frac{\alpha}{2} \right)^{\frac{1}{20}}, 1 \right]
	\]
	for example with $\alpha = 0.05$ is the CI given by $[0.83, 1]$ which is a good estimate for high $p$.
	
\end{example}
