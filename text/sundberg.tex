
\chapter{CI for Exponentially Distributed (Censored) Data}


\cite{Sundberg2001} discusses several confidence intervals for the rate $\theta$
of an exponential distribution under type-I-censoring. All of them are
\emph{pivot-based}: a function of the data and $\theta$ with a known
(approximate) distribution is inverted to obtain an interval for $\theta$.
They differ in the pivot and in whether its distribution is known exactly or not.
\begin{itemize}
	\item \textbf{Wald-type intervals} ($C1$, $C2$, $C3$, $C4$) use the asymptotic normality
	of the MLE
	\item \textbf{Likelihood ratio interval} ($C5$) inverts Wilks' statistic
	$2r \left( \ell (\hat \theta) - \ell ( \theta) \right)\overset{d}{\longrightarrow}\chi^2_1$.
	\item \textbf{$\chi^2$-based intervals} ($C6$, $C7$) use
	$2\theta\,TTT\sim\chi^2_{2r}$.
\end{itemize}

First, calculate the MLE of the parameter $\theta$ and the Fisher information since we will need those values to construct a CI for $\theta$. For this we distinguish between the different censoring schemes. 

\paragraph{Uncensored:}

The likelihood function is given by
\[
L(\theta) = \prod_{i=1}^n \theta \exp(- \theta x_i)
= \theta^n \exp\!\Big(- \theta \sum_{i = 1}^n x_i\Big),
\]
and the log-likelihood function is
\begin{equation} \label{log likelihood function uncensored}
	\ell(\theta) = n \ln (\theta) - \theta \sum_{i = 1}^n x_i .
\end{equation}
Taking the first and second derivative with respect to $\theta$ gives
\[
\ell'(\theta) = \frac{n}{\theta} - \sum_{i = 1}^n x_i,
\qquad
\ell''(\theta) = - \frac{n}{\theta^2}.
\]

We obtain the MLE $\hat\theta$ by setting $\ell'(\theta) = 0$ and solving for $\theta$
\begin{equation} \label{MLE for uncensored}
	\hat\theta = \frac{n}{\sum_{i = 1}^n x_i}. 
\end{equation}
Since $\ell''(\theta) = -n/\theta^2 < 0$ for all $\theta > 0$, the log-likelihood function is strictly concave, so $\hat\theta$ is indeed the unique maximum.

The Fisher information is defined as the expected value of the negative second derivative of the log-likelihood function,
\[
\mathcal{I}(\theta) = \mathbb{E}\left[ - \ell''(\theta) \right].
\]
In this case, $-\ell''(\theta) = n/\theta^2$ does not depend on the data $x_1, \ldots, x_n$. 
Hence it is not random, and its expected value is the constant itself
\begin{equation} \label{fisher info exp uncensored}
	\mathcal{I}_{\text{uncensored}}(\theta)
	=
	\mathbb{E} \left[ \frac{n}{\theta^2} \right]
	=
	\frac{n}{\theta^2}.
\end{equation}

We will use $\hat\theta$ and $\mathcal{I}(\theta)$ later to construct confidence intervals.

\paragraph{Type-II censored data}
Surprisingly, type-II censored data leads to results that are similar to those for uncensored data.
Let $r \in \{1, \ldots, n\}$ be fixed and let $Y_1, \ldots, Y_n$ be the type-II censored data corresponding to $X_1, \ldots, X_n$.

The likelihood function is
\begin{align*}
	L(\theta) 
	&=
	\underbrace{\prod_{i=1}^r \theta \exp(- \theta x_{i:n})}_{\text{uncensored}} \cdot \underbrace{\exp(- \theta x_{r:n})^{n-r}}_{\text{censored}} \\
	&=
	\theta^r \exp \Bigg( - \theta \bigg[ \underbrace{\sum_{i = 1}^r x_{i:n} + (n-r)\, x_{r:n}}_{=:TTT} \bigg] \Bigg),
\end{align*}
and the log-likelihood function and its derivatives are
\begin{align}
	\ell(\theta) &= r \log (\theta) - \theta \cdot TTT, \label{log likelihood function type II censored} \\
	\ell'(\theta) &= \frac{r}{\theta} - TTT, \notag \\
	\ell''(\theta) &= - \frac{r}{\theta^2}. \notag
\end{align}
Setting $\ell'(\theta) = 0$ gives the MLE
\begin{equation}  \label{MLE for type II censored}
	\hat\theta = \frac{r}{TTT}.
\end{equation}
Again, the second derivative does not depend on the data, so the Fisher information is
\begin{equation} \label{fisher info exp type II}
	\mathcal{I}_{\text{type II}}(\theta) = \mathbb{E}\left[ - \ell''(\theta) \right] = \frac{r}{\theta^2}.
\end{equation}

These are the same formulas as in the uncensored case with the change that
the sample size $n$ is replaced by the number $r$ of observed failures, and the sum of the observations $\sum_{i=1}^n x_i$ is replaced by the sum of the censored observations, also called the total time on test
\[
TTT = \sum_{i = 1}^n y_i = \sum_{i = 1}^r x_{i:n} + (n-r)\, x_{r:n}.
\]
For $r = n$ we have $TTT = \sum_{i=1}^n x_i$, so the uncensored case is included as a special case.


\paragraph{Type-I censored data}

Let $c > 0$ be a fixed time at which the experiment is stopped.
Again, we start by computing the likelihood function
\begin{align*}
	L(\theta) 
	&=
	\underbrace{\prod_{i=1}^R \theta \exp(- \theta x_{i:n})}_{\text{uncensored}} \cdot \underbrace{\exp(- \theta c)^{n-R}}_{\text{censored}} \\
	&=
	\theta^R \exp \Bigg( - \theta \bigg[ \underbrace{\sum_{i = 1}^R x_{i:n} + (n-R)\,c}_{=:TTT} \bigg] \Bigg),
\end{align*}
where $R$ is the number of uncensored observations, i.e.\ the number of $X_i$ with $X_i \leq c$.
In contrast to type-II censoring, $R$ is not fixed but a random variable.
Each observation $X_i$ is uncensored with probability
\[
p = \mathbb{P}(X_i \leq c) = 1 - \exp(-\theta c),
\]
and the observations are independent. 
Hence $R$ is binomially distributed
\[
R \sim \text{Bin}\big(n,\, 1 - \exp(-\theta c)\big).
\]

The log-likelihood function and its derivatives are given by
\begin{align}
	\ell(\theta) &= R \log \theta - \theta TTT, \label{log likelihood type I censored} \\
	\ell'(\theta) &= \frac{R}{\theta} - TTT, \notag \\
	\ell''(\theta) &= - \frac{R}{\theta^2}. \notag
\end{align}
For $R \geq 1$, setting $\ell'(\theta) = 0$ gives the MLE
\[
\hat\theta = \frac{R}{TTT},
\]
where $TTT$ depends on the data and on the random variable $R$.
For $R = 0$ we have $\ell'(\theta) = -TTT = -nc < 0$ for all $\theta$, so $\ell$ is strictly decreasing and the MLE does not exist.

This time, the Fisher information is not a constant, because the random variable $R$ appears in the second derivative. But we can take expectation of $R$ easily since we know its distribution
\[
\mathcal{I}_{\text{type I}}(\theta) 
= 
\mathbb{E} \left[ - \ell''(\theta) \right]
=
\mathbb{E} \left[ \frac{R}{\theta^2} \right]
=
\frac{\mathbb{E}[R]}{\theta^2} 
=
\frac{n \left( 1 - \exp(-c \theta) \right)}{\theta^2}.
\]

Compared to the uncensored case, the Fisher information has the extra factor $1 - \exp(-c\theta)$.
This is exactly the probability that an observation is uncensored.
Equivalently, the sample size $n$ is replaced by the expected number of observed failures $\mathbb{E}[R]$, as $r$ replaced $n$ in the type-II case.
The factor shows how much information is lost by censoring. The effect of the censoring time $c$ on the Fisher information is the following.

\begin{itemize}
	\item For $c \to \infty$, we have $1 - \exp(-c\theta) \to 1$ and we recover the Fisher information $n/\theta^2$ of the uncensored case.
	\item For $c \to 0$, we have $1 - \exp(-c\theta) \to 0$, so almost all observations are censored and the data contain almost no information about $\theta$.
\end{itemize}

All in all, the Fisher information for the rate parameter $\theta$ is given by 
\begin{equation*}
	\mathcal I (\theta) = 
	\begin{cases}
		\frac{n}{\theta^2} &\text{ for uncensored} \\
		\frac{r}{\theta^2} &\text{ for type II censoring} \\
		\frac{n(1 - \exp(- c \theta))}{\theta^2} &\text{ for type I censoring} \\
	\end{cases}
\end{equation*}

In Sections~\ref{label}--\ref{label}, seven different methods for obtaining a two-sided CI for the rate parameter $\theta$ are discussed. These methods were introduced by \cite{Sundberg2001} as $C1$--$C7$ and were presented as CIs for Type-I censored data. However, from a theoretical perspective, the CIs proposed by \cite{Sundberg2001} are mathematically correct only for uncensored and Type-II censored data, despite the author's claim that they are applicable to Type-I censored data.

The following sections provide detailed derivations of the CIs under the different censoring schemes, distinguishing between Type-I and Type-II censoring. The uncensored case does not require separate discussion, as it is a special case of Type-II censoring obtained by setting $r=n$.

Note that in this work the rate parameter is denoted by $\theta$, whereas in \cite{Sundberg2001}, $\theta$ denotes the scale parameter. This difference in parametrisation explains why the CIs appear different. However, the corresponding CIs can be obtained by taking the reciprocals of the bounds and reversing their order.

For Sections~\ref{label}--\ref{label}, let $r \in \{1,\ldots,n\}$ and $c>0$ denote fixed values defining the respective censoring schemes.


\section{C1}

\textbf{Type-II-censored data}
By the asymptotic normality of the MLE $\hat\theta$, it holds that
\begin{equation} \label{fisher info exp}
	\sqrt{\mathcal{I}(\theta)} \left( \hat\theta - \theta \right) 
	\overset{d}{\longrightarrow}
	\mathcal{N}(0, 1).
\end{equation}
The MLE is consistent, i.e.\ $\hat\theta \overset{P}{\longrightarrow} \theta$ \textcolor{red}{($\star$)}.
The mapping $g(t) = \theta / t$ is continuous on $(0, \infty)$ and with the continuous mapping theorem holds
\begin{equation} \label{CMT C1 type II}
	\hat\theta \overset{P}{\longrightarrow} \theta 
	\overset{CMT}{\implies }
	g(\hat\theta) \overset{P}{\longrightarrow} g(\theta)
	\implies
	\frac{\theta}{\hat \theta} 
	\overset{P}{\longrightarrow} \frac{\theta}{ \theta} = 1.
\end{equation}
We can plug in the Fisher information derived in \eqref{fisher info exp type II} and conclude that 
\begin{equation}
	\sqrt{\frac{\mathcal I (\hat \theta)}{\mathcal I (\theta)} }
	=
	\sqrt{\frac{r / \hat \theta^2}{r / \theta^2}}
	=
	\frac{\theta}{\hat \theta} 
	\overset{P}{\longrightarrow}
	1.
\end{equation}
Together with~\eqref{fisher info exp}, Slutsky's theorem gives
\begin{equation} \label{distn for C1}
	\sqrt{\mathcal{I}(\hat\theta)} \left( \hat\theta - \theta \right)
	=
	\sqrt{\frac{\mathcal{I}(\hat\theta)}{\mathcal{I}(\theta)}}
	\cdot
	\sqrt{\mathcal{I}(\theta)} \left( \hat\theta - \theta \right)
	\overset{d}{\longrightarrow}
	1 \cdot \mathcal{N}(0, 1).
\end{equation}

With the knowledge of the asymptotic distribution of the standardized MLE in~\eqref{distn for C1}, one can construct an approximate CI.
Let $z_p$ denote the $p$-quantile of the standard normal distribution. We directly utilize the fact that $z_{\alpha/2} = -z_{1-\alpha/2}$. 
For large $r$ it holds that
\begin{align*}
	1 - \alpha 
	&\approx
	\mathbb{P} \left( z_{\alpha / 2} \leq \sqrt{\mathcal{I}_{\text{type II}}(\hat\theta)} \left( \hat\theta - \theta \right) 
	\leq z_{1 - \alpha / 2} \right) \\
	&=
	\mathbb{P} \left( -z_{1 - \alpha / 2} \leq \sqrt{r}\, \frac{\hat\theta - \theta}{\hat\theta}  
	\leq z_{1 - \alpha / 2} \right)
\intertext{and rearranging for $\theta$ gives}
	&=
	\mathbb{P} \left( \hat\theta \left( 1 - \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right) \leq \theta \leq \hat\theta \left( 1 + \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right) \right).
\end{align*}
Hence an approximate $(1 - \alpha)$-confidence interval for $\theta$ is given by
\begin{equation*}
	C1^{\text{type II}}_{1 - \alpha} = \left[\, \hat\theta \left( 1 - \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right),\; \hat\theta \left( 1 + \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right) \right].
\end{equation*}

In \cite{Sundberg2001}, the CI $C1^{\text{type II}}_{1 - \alpha}$ is claimed to be a CI for type-I censored data, although it is a CI for type-II censored data.

\textbf{Type-I-censored data}
We again use the asymptotic normality of the MLE~\eqref{fisher info exp} and the continuous mapping theorem. Define the continuous mapping $g(t) = \frac{\sqrt{1 - \exp(-ct)}}{\sqrt{1 - \exp(-c \theta)}} \frac{\theta}{t}$. Then we can similarly to \eqref{CMT C1 type II}, conclude that
\begin{equation*}
	\hat \theta 
	\overset{P}{\rightarrow}
	\theta
	\overset{CMT}{\implies}
	g(\hat \theta) 
	\overset{P}{\rightarrow}
	g(\theta)
	\implies
	\frac{\sqrt{1 - \exp(-c \hat \theta)}}{\sqrt{1 - \exp(-c \theta)}} \frac{\theta}{\hat \theta}
	\overset{P}{\rightarrow}
	1.
\end{equation*}  
So we know that 
\begin{equation*}
	\sqrt{\frac{\mathcal I_{\text{type I}} (\hat \theta)}{\mathcal I_{\text{type I}} (\theta)}} 
	=
	\sqrt{\frac{
			\frac{n(1 - \exp(-c \hat \theta))}{\hat \theta^2}	
	}{
			\frac{n(1 - \exp(-c \theta))}{\theta^2}
	}}
	=
	\frac{\sqrt{1 - \exp(-c \hat \theta)}}{\sqrt{1 - \exp(-c \theta)}} \frac{\theta}{\hat \theta} 
	\overset{P}{\rightarrow}
	1.
\end{equation*}
So we have the same result as in \eqref{distn for C1}. With this knowledge, we can derive
\begin{align*}
	1 - \alpha 
	&\approx
	\mathbb{P} \left( 
	z_{\alpha / 2} \leq \sqrt{\mathcal I (\hat \theta)} (\hat \theta - \theta)
	\leq 
	z_{1 - \alpha / 2} 
	\right) \\
	&=
	\mathbb{P} \left( 
	z_{\alpha / 2} \leq 
	\sqrt{n (1 - \exp(-c \hat \theta))} \frac{\hat \theta - \theta}{\hat \theta} 
	\leq 
	z_{1 - \alpha / 2} 
	\right) \\
\intertext{and solve it algebraically for $\theta$}
	&=
	\mathbb{P} \left( 
	\hat \theta \left( 1 - \frac{z_{1 - \alpha / 2}}{\sqrt{n (1 - \exp(-c \hat \theta))}} \right)
	\leq
	\theta
	\leq
	\hat \theta \left( 1 + \frac{z_{1 - \alpha / 2}}{\sqrt{n (1 - \exp(-c \hat \theta))}} \right)
	\right).
\end{align*}
And this gives the CI
\begin{equation*}
	C1^{\text{type I}}_{1 - \alpha}
	\left[
	\hat \theta \left( 1 - \frac{z_{1 - \alpha / 2}}{\sqrt{n (1 - \exp(-c \hat \theta))}} \right), 
	\hat \theta \left( 1 + \frac{z_{1 - \alpha / 2}}{\sqrt{n (1 - \exp(-c \hat \theta))}} \right)
	\right]
\end{equation*}
Note that, in contrast to $C1^{\text{type II}}_{1-\alpha}$, only $r$ is replaced by the expected number of uncensored observations $n(1-\exp(-c\hat{\theta}))$. Only $C1^{\text{type II}}_{1-\alpha}$ can be expressed in the form $[c_l\hat{\theta}, c_u\hat{\theta}]$ where $c_l$ and $c_u$ are independent of the MLE $\hat{\theta}$. In contrast, $C1^{\text{type I}}_{1-\alpha}$ contains the MLE in the exponential term, making it impossible to express the confidence interval in this form.

Figure~\ref{fig:C1 comparison type I II} compares the non coverage probabilities and interval widths of $C1^{\text{type I}}_{1-\alpha}$ and $C1^{\text{type II}}_{1-\alpha}$. Note that both CIs are applied to Type-I-censored data, as \cite{Sundberg2001} claims that $C1^{\text{type II}}_{1-\alpha}$ also is a $(1-\alpha)$-CI for Type-I-censored data.

The plots in Figure~\ref{fig:C1 comparison type I II} were generated using Listing \textcolor{red}{($\star$) ref Sundberg/compare.py}. It can be observed that, for smaller values of $n$, the width of $C1^{\text{type I}}_{1-\alpha}$ is smaller than that of $C1^{\text{type II}}_{1-\alpha}$. However, the widths converge as $n$ increases.

\section{C2}

\textbf{Type-II-censored data}
We can also directly use the resulting distribution from the asymptotic normality of the MLE presented in~\eqref{fisher info exp} and use it to derive

\begin{align*}
	1 - \alpha
	&\approx
	\mathbb{P} \left( z_{\alpha / 2} \leq \sqrt{\mathcal{I}_{\text{type II}}(\theta)} \left( \hat\theta - \theta \right) 
	\leq z_{1 - \alpha / 2} \right) \\
	&=
	\mathbb{P} \left( -z_{1 - \alpha / 2} \leq \sqrt{r}\, \frac{\hat\theta - \theta}{\theta}  
	\leq z_{1 - \alpha / 2} \right)
\intertext{and rearranging gives} 
	&=
	\mathbb{P} \left( 
	\frac{\hat \theta}{1 - \frac{z_{1 + \alpha / 2}}{\sqrt{r}}}
	\leq 
	\theta
	\leq 
	\frac{\hat \theta}{1 + \frac{z_{1 - \alpha / 2}}{\sqrt{r}}}
	\right).
\end{align*}
So an approximate $(1-\alpha)$-CI  is given by
\begin{equation*}
	C2^{\text{type II}}_{1 - \alpha} = \left[
	\frac{\hat \theta}{1 + \frac{z_{1 - \alpha / 2}}{\sqrt{r}}}, 
	\frac{\hat \theta}{1 - \frac{z_{1 - \alpha / 2}}{\sqrt{r}}}
	\right].
\end{equation*}

This CI can again be written as $[c_l \hat \theta, c_u \hat \theta]$ with $c_l$ and $c_u$ independent of the MLE $\hat \theta$. 
Specifically, it holds 
\[
	c_l = \frac{1}{1 + \frac{z_{1 - \alpha / 2}}{\sqrt{r}}}, 
	\quad 
	c_u = \frac{1}{1 - \frac{z_{1 - \alpha / 2}}{\sqrt{r}}}
\]
so the constants only depend on $r$ and $\alpha$.
For certain small $r$, the interval $C2^{\text{type II}}_{1-\alpha}$ has a negative upper bound and a positive lower bound. In this case, it is not a valid interval. This can be seen in Figure~\ref{fig:sundberg_alpha_comparison}, which plots $r$ against the quantity $c_u - c_l$. This quantity is proportional to the width of the interval. The exact width is obtained by multiplying $c_u - c_l$ by $\hat\theta$.

The range of small $r$ for which $c_u - c_l$ is negative depends on $\alpha$. 
For $\alpha = 0.01$, the interval is invalid for $r \leq 6$, as shown in Figure~\ref{fig:sundberg_alpha_001}.
For $\alpha = 0.1$, the interval is invalid for $r \leq 2$, as shown in Figure~\ref{fig:sundberg_alpha_01}. 

It holds 
\[
\frac{z_{1- \alpha / 2}}{\sqrt{r}} > 1
\quad \text{if and only if} \quad
r < z_{1-\alpha/2}^2
\]
and exactly in this case, the denominator of the upper bound $1 - \frac {z_{1-\alpha/2}}{\sqrt{r}}$ is negative, while the lower bound is always positive for all $r \in \mathbb N _0$ and $\alpha \in (0, 1)$. 
So the confidence interval $C2^{\text{type II}}_{1-\alpha}$ is invalid if $r < z_{1 - \alpha / 2}^2$.
The range of $r$ for which $C2_{1 - \alpha}^{\text{type II}}$ is a valid CI changes with $\alpha$ because the critical value $z_{1-\alpha/2}^2$ increases as $\alpha$ decreases.
For example, when $\alpha = 0.1$, the interval is invalid for all
\[
n < z_{0.95}^2 \approx 2.706.
\]
When $\alpha = 0.01$, the interval is invalid for all
\[
n < z_{0.995}^2 \approx 6.635.
\]
Thus, a smaller significance level results in a larger range of sample sizes for which the interval is invalid. This is consistent with the observations shown in Figure~\ref{fig:sundberg_alpha_comparison}.

This is an important observation that should be kept in mind. However, this issue occurs only for very small values of $r$.

\textbf{Type-I-censored data}
We again use the approach from \eqref{fisher info exp}
\[
	\sqrt{\mathcal I (\theta)} (\theta - \hat \theta) = \sqrt{n (1 - \exp(-c \theta))} \frac{\hat \theta - \theta}{\theta}
	\overset{d}{\rightarrow}
\]
and solve the inequality from 
\[
	1 - \alpha 
	\approx
	\mathbb{P} \left( z_{\alpha / 2} \leq \sqrt{n (1 - \exp(-c \theta))} \frac{\hat \theta - \theta}{\theta} \leq z_{1 - \alpha / 2} \right).
\]
For this we set 
\[
	Q(t) = \sqrt{n (1 - \exp(-c t))} \frac{\hat \theta - t}{t}
\]
and notice that $Q$ is strictly decreasing in $t$. \section{C3}

For C3 and C4, \cite{Sundberg2001} uses the Delta method to obtain a standard normal limit distribution.

\textbf{Type-II-censored data}
To construct $C3_{1-\alpha}^{\text{type II}}$, we first rearrange~\eqref{fisher info exp} to obtain
\begin{equation} \label{Sundberg C3 distn}
	\sqrt{\mathcal{I}_{\text{type II}}(\theta)} \left( \hat\theta - \theta \right)
	=
	\frac{\sqrt{r}}{\theta} \left( \hat\theta - \theta \right)
	=
	\sqrt{r} \Big( \underbrace{\frac{\hat \theta}{\theta}}_{=: Z} - 1 \Big)
\end{equation}
and by~\eqref{fisher info exp} this converges in distribution, i.e. $\sqrt{r} \left( Z - 1 \right) \overset{d}{\longrightarrow} \mathcal{N}(0, 1)$.

We now apply the Delta method with the function $g(x) = \log x$.
It holds that $g'(1) = 1 \neq 0$, so the conditions of the Delta method are satisfied, and we obtain
\begin{equation} \label{C3 Delta method Sundberg}
	\sqrt{r} \left( g(Z) - g(1) \right) \overset{d}{\longrightarrow} \mathcal{N} \left( 0, \, |g'(1)|^2 \right).
\end{equation}
Since $g(Z) - g(1) = \log(\hat\theta / \theta) - \log 1 = \log \hat\theta - \log \theta$ and $|g'(1)|^2 = 1$, this simplifies to
\begin{equation*}
	\sqrt{r} \left( \log \hat\theta - \log \theta \right) \overset{d}{\longrightarrow} \mathcal{N}(0, 1).
\end{equation*}
This is the distribution used to construct C3 in~\cite{Sundberg2001}. For sufficiently large $r$ holds
\begin{align*}
	1 - \alpha 
	\approx&
	\mathbb{P} \left( z_{\alpha / 2} \leq \sqrt{r} \left( \log \hat\theta - \log \theta \right) \leq z_{1 - \alpha / 2} \right)
\intertext{and rearranging gives}
	=&
	\mathbb{P} 
	\left( 
	\hat \theta  \cdot \exp \left( - \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right)
	\leq \theta \leq 
	\hat \theta \cdot \exp \left( \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right)
	\right).
\end{align*}
So another $(1 - \alpha)$-CI is given by 
\begin{equation*}
	C3_{1 - \alpha}^{\text{type II}} = 
	\left[
	\hat \theta  \cdot \exp \left( - \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right)
	,
	\hat \theta \cdot \exp \left( \frac{z_{1 - \alpha / 2}}{\sqrt{r}} \right)
	\right].
\end{equation*}


\textbf{Type-I-censored data}

For Type-I-censored data, we obtain
\[
\sqrt{\mathcal I_{\text{type I}}(\theta)} \left( \hat \theta - \theta \right)
=
\sqrt{n (1 - \exp(-c \theta))} \left( \frac{\hat \theta}{\theta} - 1 \right).
\]

As in the Type-II censoring case, the Delta method is applied with the function $g(x) = \log(x)$. Since $g'(1) = 1$, it follows that
\[
\sqrt{n (1 - \exp(-c \theta))}
\left(
g\left(\frac{\hat \theta}{\theta}\right)-g(1)
\right)
\overset{d}{\longrightarrow}
\mathcal{N}(0,1)
\]
and using the logarithmic properties, this expression can be rewritten as
\[
\sqrt{n (1 - \exp(-c \theta))}
\left(
\log(\hat \theta)-\log(\theta)
\right)
\overset{d}{\longrightarrow}
\mathcal{N}(0,1).
\]

Thus, for sufficiently large $n$, an approximate $(1-\alpha)$-CI can be obtained by finding the solution of the inequality in
\[
\begin{aligned}
	1-\alpha \approx
	\mathbb{P}\Bigg(
	z_{\alpha/2}
	\leq {}&
	\sqrt{n (1-\exp(-c\theta))}
	\left(\log(\hat\theta)-\log(\theta)\right)
	\leq & z_{1-\alpha/2}
	\Bigg).
\end{aligned}
\]

As in $C2$, the resulting inequality of the Type-II censoring case cannot be rearranged explicitly to obtain the lower and upper bounds, since the factor $\sqrt{n(1-\exp(-c\theta))}$ also depends on the unknown parameter $\theta$. Therefore, the confidence bounds must be determined numerically.

The function \texttt{C3()} implemented in \texttt{Sundberg/sundberg\_type\_I.py} numerically determines the solutions of the corresponding boundary equations and returns $C3_{1-\alpha}^{\text{type I}}$.

\section{C4}

\textbf{Type-II-censored data}
Again, we will use the Delta method but this time with the mapping $g(x) = x^{-1/3}$. 
Again, we will use the asymptotic normality of the MLE~\eqref{Sundberg C3 distn}.
It holds $g(Z) =\left(  \frac{ \theta}{\hat\theta} \right)^{\frac{1}{3}}, g(1) = 1$ and $|g'(1)|^2 = \frac{1}{9}$. If we plug in these realisations of $g$ into \eqref{C3 Delta method Sundberg} we obtain
\begin{align*}
	\sqrt{r} \left( \frac{\theta^{1 / 3} - \hat \theta^{1 / 3}}{\hat \theta^{1 / 3}} \right) \overset{d}{\rightarrow} \mathcal N \left( 0, \frac{1}{9} \right).
\end{align*}
To get the standard normal distribution, one can multiply by $3$ what gives us
\begin{align}
	3 \sqrt{r} \left( \frac{\theta^{1 / 3} - \hat \theta^{1 / 3}}{\hat \theta^{1 / 3}} \right) \overset{d}{\rightarrow} \mathcal N \left( 0, 1 \right). \label{distn for C4}
\end{align}

Using \eqref{distn for C4} gives for large $r$
\begin{align}
	1 - \alpha 
	&\approx
	\mathbb{P} 
	\left( 
	z_{\alpha / 2} 
	\leq
	3 \sqrt{r} \left( \frac{\theta^{1 / 3} - \hat \theta^{1 / 3}}{\theta^{1 / 3}} \right)
	\leq
	z_{1 - \alpha / 2}
	\right)
	\intertext{what can be rearranged to}
	&=
	\mathbb{P} 
	\left( 
	\left( 1 - \frac{z_{1 - \alpha / 2}}{3\sqrt{r}} \right)^3 \cdot \hat \theta
	\leq
	\theta
	\leq
	\left( 1 + \frac{z_{1 - \alpha / 2}}{3\sqrt{r}} \right)^3 \cdot \hat \theta
	\right)
\end{align}
All in all, we get the approximate CI
\[
C4^{\text{type II}}_{1-\alpha}
=
\left[
\left( 1 - \frac{z_{1 - \alpha / 2}}{3\sqrt{r}} \right)^3 \cdot \hat \theta 
,
\left( 1 + \frac{z_{1 - \alpha / 2}}{3\sqrt{r}} \right)^3 \cdot \hat \theta
\right].
\]

\textbf{Type-I-censored data}
Similarly to the construction of $C3^{\text{type II}}_{1-\alpha}$, we can derive the asymptotiv distribution 
\begin{equation}
	3 \sqrt{n (1 - \exp(-c \theta))} \frac{\theta^{1/3} - \hat \theta^{1/3}}{\theta^{1/3}}
	\overset{d}{\longrightarrow}
	\mathcal N (0, 1)
\end{equation}
which we can solve numerically. Find an implementation in \texttt{Sundberg/sundberg\_type\_I.py} in the function \texttt{C4()}.

\section{C5}

\cite{Sundberg2001} presents with C5 a likelihood based CI.

\textbf{Type-II-censored data}
The log-likelihood function for Type-II-censored data is given by~\eqref{log likelihood function type II censored}.
According to Wilks' theorem (for details about Wilks Theorem see Theorem 10.3.1 in \cite{casella2024statistical}), the likelihood ratio statistic converges in distribution to a $\chi^2_1$ distribution as the sample size increases, i. e. 
\begin{equation} \label{Wilks Theorem}
	2\left(\ell(\hat{\theta})-\ell(\theta)\right)
	\overset{d}{\longrightarrow}
	\chi^2_1
\end{equation}
Using basic calculus and substituting the MLE, \(\hat{\theta}=r/TTT\), as derived in~\eqref{MLE for type II censored}, we obtain
\begin{align} \label{log likelihood wilk}
	2\left(\ell(\hat{\theta})-\ell(\theta)\right)
	=
	2r\left(
	\frac{\theta}{\hat{\theta}}-1
	-\log\left(\frac{\theta}{\hat{\theta}}\right)
	\right) 
	\overset{d}{\longrightarrow}
	\chi^2_1
	.
\end{align}

Therefore, for sufficiently large \(r\), we have
\begin{align} \label{C5 prob}
	1-\alpha \approx
	\mathbb{P}\left(
	2r  \left(
	\frac{\theta}{\hat\theta}-1
	-\log\left(\frac{\theta}{\hat\theta}\right)
	\right)
	\leq \chi^2_{1,1-\alpha}
	\right),
\end{align}
where \(\chi^2_{1,1-\alpha}\) denotes the \((1-\alpha)\)-quantile of the $\chi^2_1$ distribution.

We can use the well-known identity
\[
\chi^2_{1,1-\alpha}=z_{1-\alpha/2}^2
\]
instead. This identity allows us to rewrite the inequality observed in~\eqref{C5 prob} as
\begin{equation} \label{inequality C5}
	2 r \left(
	\frac{\theta}{\hat{\theta}}-1
	-\log\left(\frac{\theta}{\hat{\theta}}\right)
	\right)
	\leq z_{1-\alpha/2}^2.
\end{equation}
The CI can be obtained by solving this inequality for \(\theta\).
To solve the inequality, we define the function
\[
h(x)=x-1-\log(x).
\]
for $x \in (0, \infty)$.
Furthermore, for simplicity, we set \(t^*=\theta/\hat{\theta}\). With this notation,~\eqref{inequality C5} becomes
\begin{equation} \label{inequality C5 h}
	h(t^*) \leq \frac{z_{1-\alpha/2}^2}{2r}.
\end{equation}

As we can see in Figure~\ref{fig:h function C5}, the function \(h\) is strictly convex and has a unique global minimum at \(1\), where \(h(1)=0\). Moreover, \(h\) tends to infinity as \(x\) approaches $0$ or infinity. Consequently, for every constant \(K>0\), the equation \(h(x)=K\) has exactly two solutions, one in the interval \((0,1)\) and the other in \((1,\infty)\).
These two solutions can be expressed using the Lambert \(W\) function. More precisely, the inverse functions on the two intervals are given by
\begin{equation} \label{inverse of h}
	h^{-1}_{|(0,1)}(y)
	=
	-W_0\left(-\exp(-y-1)\right)
	\quad \text{and} \quad
	h^{-1}_{|(1,\infty)}(y)
	=
	-W_{-1}\left(-\exp(-y-1)\right),
\end{equation}
where \(W_0\) and \(W_{-1}\) denote the principal and the lower real branch of the Lambert \(W\) function.
The values of the Lambert \(W\) function can be calculated numerically in Python using \texttt{scipy.special.lambertw}.
For more details about the Lambertz W function see Appendix \textcolor{red}{($\star$)} or \cite{veberic2010having}.
Applying these inverse functions to~\eqref{inequality C5 h}, we obtain
\[
t^* \in
\left[
h^{-1}_{|(0,1)}
\left(\frac{z_{1-\alpha/2}^2}{2r}\right),
\;
h^{-1}_{|(1,\infty)}
\left(\frac{z_{1-\alpha/2}^2}{2r}\right)
\right].
\]

Finally, we define the lower and upper constants as
\begin{align} \label{C5 constants}
	c_l =
	h^{-1}_{|(0,1)}
	\left(\frac{z_{1-\alpha/2}^2}{2r}\right)
	\quad \text{and} \quad
	c_u =
	h^{-1}_{|(1,\infty)}
	\left(\frac{z_{1-\alpha/2}^2}{2r}\right).
\end{align}
Thus, we have
\[
c_l \leq \frac{\theta}{\hat{\theta}} \leq c_u.
\]
Since the MLE \(\hat{\theta}\) is positive, we can multiply all parts of the inequality by \(\hat{\theta}\) without changing the direction of the inequalities. Therefore, the resulting confidence interval for \(\theta\) is
\begin{equation*}
	C5_{1-\alpha}^{\text{type II}}
	=
	[c_l\hat{\theta},\,c_u\hat{\theta}]
\end{equation*}
with $c_l$ and $c_u$ defined as in \eqref{C5 constants}.
It holds $c_l < 1$ because the left bound of $C5_{1-\alpha}^{\text{type II}}$ is supposed to be smaller than the MLE $\hat \theta$ and the other constant $c_u > 1$ is greater than 1, such that the right bound is greater than $\hat \theta$. One can visualize that $c_l \in (0, 1)$ and $c_u \in (1, \infty)$ in Figure~\ref{fig:h function C5}.

\begin{figure}[htbp]
	\centering	
	\includegraphics[width=0.5\textwidth]{figures/Sundberg_figures/sundberg_C5/plot_h_for_C5}
	\caption{Function $h: (0, \infty)  \to [0, \infty), h(x)= x - 1- \log(x)$ used to solve inequality~\eqref{inequality C5}. If we fix a constant $K > 0$, then the interval $[c_l, c_u]$ has all solution to the inequality $h(x)\leq K$.}
	\label{fig:h function C5}
\end{figure}



\textbf{Type-I-censored data}

The regularity conditions for Wilks Theorem are also fulfilled for the log likelihood function~\eqref{log likelihood type I censored} and with \eqref{Wilks Theorem}, we can derive 
\begin{equation}
	2 R \left( \frac{\theta}{\hat \theta} - 1- \log \left( \frac{\theta}{\hat \theta} \right) \right) 
	\overset{d}{\longrightarrow}
	\chi^2_1.
\end{equation}

The random variable $R$ and the MLE $\hat \theta$ are both functions of the data and we can just plug in the observed values. We get the same CI as in the Type-II case but now $r$ depends on the data, i. e. 
\begin{equation*}
	C5_{1-\alpha}^{\text{type I}}
	=
	[c_l\hat{\theta},\,c_u\hat{\theta}]
\end{equation*}
with $c_l, c_u$ as in~\eqref{C5 constants}, but use $R = \sum_{i = 1}^n \mathds{1}(X_i < c)$ instead of the fixed value $r$.
Note that this is exactly what \cite{Sundberg2001} proposed, so the results of 
\texttt{sundberg.C5()} and \texttt{sundberg\_type\_I.C5()} will deliver the same results.


\section{C6}


\textbf{Type-II-censored data}
Define the normalized spacings for $i \in \{0, \ldots, n-1\}$ as
\[
D_i := (n-i)\left( X_{(i+1):n} - X_{i:n} \right)
\]
where we set $X_{0:n} = 0$.
By Theorem~3.1.1 in~\cite{galambos2006characterizations}, the normalized spacings $D_0, \ldots, D_{n-1}$ are independent identically $\text{Exp}(\theta)$ distributed.
Furthermore, an index shift in the sums shows that
\[
TTT = \sum_{i=0}^{r-1} D_i .
\]
Since a sum of independent exponentially distributed random variables is Gamma distributed (see Example C1.16. (i) in \cite{cramer2008grundlagen}), we obtain
\[
TTT \sim \Gamma(r, \theta).
\]
Scaling by $2\theta$ yields
\begin{equation} \label{distn for C6}
	2\theta \cdot TTT \sim \Gamma\!\left(r, \tfrac{1}{2}\right) = \chi^2_{2r}.
\end{equation}

Because the distribution in~\eqref{distn for C6} is known exactly, it holds for every $r \geq 1$ that
\begin{align*}
	1 - \alpha
	&=
	\mathbb{P}\left( \chi^2_{2r,\,\alpha/2}
	\leq
	2\theta \cdot TTT
	\leq
	\chi^2_{2r,\,1-\alpha/2}
	\right) \\
	&=
	\mathbb{P}\left( \frac{\chi^2_{2r,\,\alpha/2}}{2 \cdot TTT}
	\leq
	\theta
	\leq
	\frac{\chi^2_{2r,\,1-\alpha/2}}{2 \cdot TTT}
	\right) \\
	\intertext{and substituting the MLE $\hat\theta = r / TTT$, we get}
	&=
	\mathbb{P}\left(
	\frac{\chi^2_{2r,\,\alpha/2}}{2r}\,\hat\theta
	\leq
	\theta
	\leq
	\frac{\chi^2_{2r,\,1-\alpha/2}}{2r}\,\hat\theta
	\right).
\end{align*}

Hence, an exact $(1-\alpha)$ confidence interval for $\theta$ is given by
\[
C6_{1-\alpha} = \left[
\frac{\chi^2_{2r,\,\alpha/2}}{2r}\,\hat\theta,\;
\frac{\chi^2_{2r,\,1-\alpha/2}}{2r}\,\hat\theta
\right].
\]


\textbf{Type-I-censored data}
For type-I-censored data the number of observed failures $R$ is random, and the statistic $2\theta\cdot TTT$ is no longer exactly $\chi^2$-distributed. 
In the type-II case the identity $TTT=\sum_{i=0}^{r-1}D_i$ holds with a fixed number of summands, whereas for type-I-censoring, we have a random number of summands.

Still, \cite{Sundberg2001} proposes to plug in the observed $R$ into the CI $C6_{1-\alpha}$ and use it for Type-I-censored data. In Figure~\ref{fig:C6 exact vs asymptotic}, one can see that $C6_{1-\alpha}$ is an exact CI for Type-II-censored data, but it also seems to be an approximate CI for Type-I-censored data.

\section{C7}
\cite{Sundberg2001} proposes to use the $\chi^2_{2r+1}$ distribution instead of the $\chi^2_{2r}$ distribution.
A heuristic motivation is the following. For type-I-censored data, the total time on test satisfies $S_R \leq TTT < S_{R+1}$ with $S_k = \sum_{i=0}^{k-1} D_i \sim \Gamma(k,\theta)$, so $2\theta\, TTT$ lies between a $\chi^2_{2R}$ and a $\chi^2_{2R+2}$ distributed random variable. Using $2r+1$ degrees of freedom corresponds to the midpoint of these two cases. This gives
\[
C7_{1 - \alpha} = \left[
\frac{\chi^2_{2r+1,\, \alpha / 2}}{2r}\, \hat \theta,\;
\frac{\chi^2_{2r+1,\, 1 - \alpha / 2}}{2r}\, \hat \theta
\right],
\]
with $r$ replaced by the observed number of failures $R$ for type-I-censored data.

For type-II-censored data, $2\theta \cdot TTT \sim \chi^2_{2r}$ holds exactly. Thus $C7_{1-\alpha}$ is no longer an exact CI for type-II-censored data. 

In Figure~\ref{fig:C7 exact vs asymptotic}, we can observe that $C7_{1 - \alpha}$ is not an exact CI for Type-II-censored data anymore, but asymptotic in both cases. On the other hand, the non-coverage probability converges faster to $\alpha$ for type-I-censored data as in Figure~\ref{fig:C6 exact vs asymptotic}.

\section{Deriving a One-Sided CI}

One-sided confidence intervals can also be constructed instead of the two-sided intervals C1--C7.

The intervals $[c_l \hat{\theta}, c_u \hat{\theta}]$ are constructed such that both tail probabilities are equal to $\alpha/2$, i.e.
\begin{align*}
	\mathbb{P} \left( \theta < c_l \hat{\theta} \right)
	=
	\mathbb{P} \left( \theta > c_u \hat{\theta} \right)
	=
	\alpha / 2.
\end{align*}

Therefore, the intervals $[c_l \hat{\theta}, \infty)$ and $(0, c_u \hat{\theta}]$ are $(1-\alpha/2)$-confidence intervals.

For example, to construct a one-sided $95\%$ CI, the corresponding bound from a two-sided $90\%$ confidence interval $[c_l \hat{\theta}, c_u \hat{\theta}]$ can be used. Thus, $[c_l \hat{\theta}, \infty)$ and $(0, c_u \hat{\theta}]$ are one-sided $95\%$ CIs.

In the file \texttt{Sundberg \slash CP\_with\_CensoPy.py}, one get set the global parameter \texttt{ci\_type = 1}, to get the non coverage probabilities from the upper and lower $(1- \alpha)$-CIs. \textcolor{red}{($\star$) See listing...}


\section{Non-Coverage Probability}


In this section, we establish a method to calculate the theoretical non-coverage probability of a CI of the form
\[
CI = [c_l \hat{\theta}, c_u \hat{\theta}]
\]
which we have in all methods from~\cite{Sundberg2001}.

The non-coverage probability of a two sided intercal can be expressed in terms of the CDF of the MLE as
\begin{align} \label{non cover prob for uncensored}
	\mathbb{P} \left( \text{non-coverage} \right)
	&=
	\mathbb{P} \left( \theta < c_l \hat{\theta} \right)
	+
	\mathbb{P} \left( \theta > c_u \hat{\theta} \right) \notag \\
	&=
	1 - F^{\hat{\theta}} \left( \frac{\theta}{c_l} \right)
	+ F^{\hat{\theta}} \left( \frac{\theta}{c_u} \right).
\end{align}

The CDF of the MLE $\hat{\theta}$ depends on the specific censoring scheme. We first consider the case without censoring and derive the corresponding CDF. For the other censoring schemes, we use the Python package \texttt{CensoPy}, developed by~\cite{Cramer2026a}. This package allows us to numerically approximate the CDF of the MLE for a wide range of censoring schemes.

We then plot the non-coverage probabilities against the sample size $n$ for different degrees of censoring and different confidence levels $\alpha$.

Figures \textcolor{red}{($\star$)} in this work correspond to Figures~2--5 in~\cite{Sundberg2001}. However, we use a different method to obtain the CDF of the MLE $\hat{\theta}$. In particular, \cite{Sundberg2001} uses a Fourier inversion of the characteristic function of the truncated exponential distribution. This method has some limitations because \textcolor{red}{($\star$)}.

In this section we will use the parametrisation $1 / \theta$ for the rate to be consistent with the parametrisation in~\cite{Sundberg2001} and \cite{Cramer2026a}. Also the package \texttt{CensoPy} uses the rate parametrisation of $1 / \theta$. Since we constructed the CIs C1 - C7 with the rate parametrisation of $\theta$, we need to adjust the CIs. 
Let $\theta' = 1 / \theta$. Then $[c_l' \hat \theta', c_u' \hat \theta']$ is a $(1 - \alpha)$ CI for $\theta'$. If we want to construct an $(1- \alpha)$ CI for $\theta$, we need to take reciprocals because 

\begin{align}
	1 - \alpha 
	&\approx
	\mathbb{P} \left(c_l' \hat \theta' \leq \theta\leq c_u' \hat \theta' \right) \\
	&=
	\mathbb{P} \left(\frac{1}{c_l' \hat \theta'} \geq \theta' \geq \frac{1}{c_u' \hat \theta'} \right) \\
\intertext{but $\hat \theta' = 1 / \hat \theta$, so}
	&=
	\mathbb{P} \left(\frac{\hat \theta}{c_l'} \geq \theta' \geq \frac{\hat \theta}{c_u'} \right)
\end{align}
So if we set $c_l = 1 / c_u'$ and $c_u = 1 / c_l'$, then $[c_l \hat \theta, c_u \hat \theta]$ is a CI for $\theta$. 

In the code, one can choose whether to use the rate parametrisation $\theta$ or the parametrisation $1/\theta$.
The function 

\texttt{CI(method\_nr, r, alpha, theta\_is\_rate)} 

in \texttt{Sundberg\slash sundberg.py} combines all seven CI methods proposed by \cite{Sundberg2001} into a single function. The argument \texttt{method\_nr} determines which method is used.
Also, one can set \texttt{theta\_is\_rate = True} to use the rate parametrisation $\theta$, or \texttt{theta\_is\_rate = False} to use the parametrisation $1/\theta$. To stay consistent with the parametrisation used in \cite{Cramer2026a}, we set \texttt{theta\_is\_rate = False} in this section.

\subsection{Theoretical Non-Coverage Probability for Non-Censored Data}

According to Model D 4.2 in \cite{cramer2008grundlagen}, the scaled MLE follows a $\chi^2$ distribution
\[
\frac{2n \hat{\theta}}{\theta} \sim \chi^2_{2n}.
\]

We can use this result to derive an expression for the CDF of the MLE
\begin{align*}
	F^{\hat{\theta}} (t)
	=
	\mathbb{P} \left( \hat{\theta} \leq t \right)
	=
	\mathbb{P} \left( \frac{2n }{\theta} \hat{\theta} \leq \frac{2n}{\theta} t \right)
	=
	F^{\chi^2_{2n}} \left( \frac{2nt}{\theta} \right).
\end{align*}

Therefore, the non-coverage probability in~\eqref{non cover prob for uncensored} can be calculated using the $\chi^2_{2n}$ distribution
\begin{equation} \label{non coverage with chi sq}
\mathbb{P} \left( \text{non-coverage} \right)
=
1
- F^{\chi^2_{2n}} \left( \frac{2n}{c_l} \right)
+ F^{\chi^2_{2n}} \left( \frac{2n}{c_u} \right).
\end{equation}

The parameter $\theta$ cancels out. Therefore, the non-coverage probability is independent of the rate parameter $1/ \theta$.
This can be easily implemented using the \texttt{chi2} distribution from \texttt{scipy.stats}, as shown in Listing~\textcolor{red}{($\star$)}.

Figures 2-5 in~\cite{Sundberg2001} are independent of the parametrisation of the exponential distribution. Indeed the censoring time $c$ is chosen dependent on $\theta$, i. e.
\begin{align}
	c = \nu \theta, \quad \nu \in \{0.5, 1, 2, \infty\}.
\end{align}

Four different values of $\nu$ are considered. These values determine the 
probability of censoring
\begin{align*}
	\mathbb{P} \left( X \text{ censored} \right)
	=
	\mathbb{P} \left( X > c \right)
	=
	\exp \left( - \frac{c}{\theta} \right)
	&=
	\exp \left( - \nu \right) \\
	&=
	\begin{cases}
		0.6065307 &\text{ for } \nu = 0.5 \\
		0.3678794 &\text{ for } \nu = 1 \\
		0.1353353 &\text{ for } \nu = 2 \\
		0 &\text{ for } \nu = \infty.
	\end{cases}
\end{align*}

Thus, as $\nu$ increases, the probability of censoring decreases. 
The case $\nu = 0.5$ represents the heaviest censoring considered in 
this work. For $\nu = \infty$, the censoring threshold is $c = \infty$, 
which means that no censoring is applied.
