\chapter{Confidence Intervals}

\begin{definition}[Confidence Interval]
Let $X = (X_1, \dots, X_n)$ be a random sample with distribution depending on an unknown parameter $\theta \in \Theta$. A \emph{confidence interval} for $\theta$ at confidence level $1 - \alpha$, $\alpha \in (0,1)$, is a pair of statistics $L(X), U(X)$ with $L(X) \leq U(X)$ such that
\[
    \mathbb{P}_\theta\big(\theta \in [L(X), U(X)]\big) \geq 1 - \alpha \quad \text{for all } \theta \in \Theta.
\]
\end{definition}

\begin{definition}[Confidence Bounds]
If only a lower bound is of interest, $[L(X), \infty)$ is called a \emph{one-sided lower confidence interval}, satisfying
\[
    \mathbb{P}_\theta\big(\theta \geq L(X)\big) \geq 1 - \alpha \quad \text{for all } \theta \in \Theta.
\]
Analogously, $(-\infty, U(X)]$ is called a \emph{one-sided upper confidence interval}.
\end{definition}



\section{Exact vs. Asymptotic CIs}

\textcolor{red}{($\star$) Exact vs Asymptotic CI, plot x = n, y = P(noncoverage), dann müsste man bei exact sehen, dass n keinen einfluss auf p(nc) hat und bei asymptotischen CI sieht man dass diese mit größeren n besser wird, gib kleines Beispiel und plots (siehe \texttt{exact\_vs\_asymptotic.R})}

\begin{definition}[Exact confidence interval]
	A confidence interval $CI$ for a parameter $\theta$ is called \textbf{exact}, if and only if
	\[
		\mathbb{P}_{\theta} \left( \theta \in CI \right) = 1 - \alpha.
	\]
\end{definition}

\begin{definition}[Asymptotic confidence interval]
	A confidence interval $CI$ for a parameter $\theta$ is called \textbf{asymptotic}, if
	\[
	\mathbb{P}_{\theta} \left( \theta \in CI \right) \overset{n \leftarrow \infty}{\rightarrow} 1 - \alpha.
	\]
	where $n$ is the sample size.
\end{definition}

So the main difference between exact and asymtotic confidence intervals is, that exact confidence intervals guarantee a confidence niveau of $1 - \alpha$ even for a small sample size $n$, but asymptotic confidence intervals only have confidence niveau for large $n$.

\begin{example}[Exact vs. Asymptotic Confidence Interval]
	Let $X_1, \ldots , X_n \overset{iid}{\sim} \mathcal N (\mu, \sigma)$. Consider two cases.
	
	
	\textbf{Case 1: standard deviation $\sigma$ is known} \\
	In this case we can construct an exact confidence interval the following way.
	Observe that the arithmetic mean of the observations fulfils $\bar X = \frac{1}{n} \sum_{i=1}^n X_i \sim \mathcal N \left( \mu, \frac{\sigma^2}{n} \right)$ and by scaling this term, we get
	\[
		\frac{\sqrt{n}(\bar X - \mu)}{\sigma} \sim \mathcal N (0, 1).
	\]
	This distribution holds for every sample size $n$ so it is not only asymptotic. 
	Let $\alpha \in (0, 1)$ be arbitrary. We can construct a confidence interval by 
	\begin{align}
		& 1 - \alpha \notag \\
		=&
		1 - \frac{\alpha}{2} - \frac{\alpha}{2} \notag \\
		=&
		\mathbb{P} \left( z_{\frac{\alpha}{2}} \leq 
		\frac{\sqrt{n}(\bar X - \mu)}{\sigma}
		\leq z_{1 - \frac{\alpha}{2}} \right) \notag \\
		=&
		\mathbb{P} 
		\left( 
		\frac{\sigma}{\sqrt{n}} z_{\frac{\alpha}{2}} \leq \bar X - \mu \leq \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}		
		\right)  \notag \\
		=&
		\mathbb{P} 
		\left( 
		\bar X - \frac{\sigma}{\sqrt{n}} z_{\frac{\alpha}{2}} \geq  \mu \geq \bar X - \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}		
		\right)
	\end{align}
	For the quantiles of the normal distributino it holds $z_p = - z_{1-p}$ for all $p \in (0, 1)$, so we can substitute the quantile $z_{\frac{\alpha}{2}}$ above. The $1- \alpha$ confidence interval is given by
	\[
		CI_{1 - \alpha}^{\text{exact}} (\mu) 
		=
		\left[ 
		\bar X - \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		,
		\bar X + \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		\right]
	\]
	We have shown that the probability that the real parameter $\mu$ is in $CI_{1 - \alpha}^{\text{exact}} (\mu)$ is $1 - \alpha$ independently of the sample size $n$, so $CI_{1 - \alpha}^{\text{exact}} (\mu)$ is an exact confidence interval. 
	
	
	\textbf{Case 2: standard deviation $\sigma$ is unknown} \\
	
	
	From the Central Limit Theorem, we know that
	\[
		\frac{\sqrt{n} (\bar X - \mu)}{\sigma} \overset{n \rightarrow \infty}{\longrightarrow} \mathcal N (0, 1).
	\]
	Since $\sigma$ is unknown, we need to substitute it somehow.
	It holds 
	\[
		S_n := \sqrt{\frac{1}{n-1} \sum_{i=1}^n \left( X_i - \bar X \right)^2 }  \overset{p}{\longrightarrow} \sigma 
	\]
	\textcolor{red}{($\star$) Warum ist $S_n$ so verteilt?}
	
	With Slutsky's Theorem, we can conclude that 
	\[
		\frac{\sqrt{n} (\bar X - \mu)}{S_n} \overset{n \rightarrow \infty}{\longrightarrow} \mathcal N (0, 1).
	\]
	and with that knowledge we can create a $1 - \alpha$ confidence interval similarily to case 1
	\[
		CI_{1 - \alpha}^{\text{asymptotic}} (\mu) 
		=
		\left[ 
		\bar X - \frac{S_n}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		,
		\bar X + \frac{S_n}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		\right].
	\]
	This confidence interval is only asymptotic because \textcolor{red}{($\star$) Warum nur asymptotic? Warum funktioniert die einsetzung mit slutsky nur asymptotic?}
	
	
	In Figure~\ref{fig:exact_vs_asymptotic}, we can see the non-coverage probability of the intervals $CI_{1 - \alpha}^{\text{exact}} (\mu)$ and $CI_{1 - \alpha}^{\text{asymptotic}} (\mu)$ dependent on the sample size. In Listing~\ref{listing:exact vs asymptotic} is the code used for generating Figure~\ref{fig:exact_vs_asymptotic}. 
	
	We can observe that \(CI_{1-\alpha}^{\mathrm{exact}}(\mu)\) maintains a non-coverage probability close to \(\alpha=0.1\) for all sample sizes $n$, while the non-coverage probability of \(CI_{1-\alpha}^{\mathrm{asymptotic}}(\mu)\) approaches \(\alpha\) as \(n\) increases.
	
	\begin{figure}[htbp]
		\centering
		\includegraphics[width=0.8\textwidth]{figures/exact\_vs\_asymptotic\_CI.png}
		\caption{Non-coverage probability of \(CI_{1-\alpha}^{\mathrm{exact}}(\mu)\) and \(CI_{1-\alpha}^{\mathrm{asymptotic}}(\mu)\) based on 1000 Monte Carlo simulations for each sample size \(n=2,\ldots,100\). For each simulation, a sample from \(N(\mu,\sigma^2)\) with \(\mu=10\) and \(\sigma=100\) is generated, and it is recorded whether \(\mu\) is contained in the respective confidence interval.}
		\label{fig:exact_vs_asymptotic}
	\end{figure}
		
\end{example}

\subsection{C1}

The MLE $\hat \theta$ of independent exponentially distributed random variables $X_1, \ldots , X_n$ is given by
\[
    \hat \theta_n = \frac{1}{\frac{1}{n} \sum_{i=1}^n  X_i}
\]

\[
    \hat \theta_n \overset{d}{\longrightarrow} \mathcal{N} (\theta, \mathcal I (\theta)^{-1})
\]

Replacing the variance by the observed variance:
\[
	\hat \theta_n \overset{d}{\longrightarrow} \mathcal{N} (\theta, \mathcal I (\hat \theta_n)^{-1})
\]

Note that this only holds if the sample size $n$ is very large.

Using this, we know
\[
    \sqrt{\mathcal I (\hat \theta_n )} \left(\hat \theta_n - \theta \right) \sim \mathcal N (0, 1)
\]


For example for $X_1, \ldots , X_n \overset{iid}{\sim} \text{Exp}(\theta)$, the Fisher information can be obtained by calculating the Likelihood function

\[
    L(\theta; \mathbf{x}) 
    =
    \prod_{i=1}^n  \theta \exp(- \theta x_i)
    =
    \theta^n \exp \left( - \theta \sum_{i=1}^n  x_i \right)
\]

\begin{align}
    l(\theta) &= n \log( \theta ) - \theta \sum_{i=1}^n  x_i  \label{log likelihood exp} \\
    l'(\theta) &= \frac{n}{\theta} - \sum_{i=1}^n  x_i \notag \\
    l''(\theta) &= - \frac{n}{\theta^2} \notag
\end{align}

So the Fisher information is 
\[
    \mathcal I (\theta) = \frac{n}{\theta^2}
\]
and therefore the MLE has distribution 

\[
    \hat \theta_n \sim \mathcal{N} \left(\theta , \frac{ \hat \theta_n^2}{n} \right)
\]

So we can get the standardized MLE 
\[
    \sqrt{n}\frac{\hat \theta_n - \theta}{\hat \theta_n} \sim \mathcal{N} \left(0 , 1 \right)
\]

With the knowledge of the distribution of the standardized MLE, one can construct an exact confidence interval. Notice that 
\begin{align*}
    \mathbb{P}
    \left(
    z_{\alpha /2} \leq \sqrt{n}\frac{\hat \theta_n - \theta}{\hat \theta_n} \leq z_{1 - \frac{\alpha}{2}}
    \right) 
    =
    1 - \alpha
\end{align*}
and by transforming the event one gets 
\begin{align*}
    \mathbb{P}
    \left(
        \hat \theta_n \left(1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}\right) 
    \leq 
        \theta 
    \leq
        \hat \theta_n \left(1 - \frac{z_{\frac{\alpha}{2}}}{\sqrt{n}} \right)
    \right) 
    =
    1 - \alpha
\end{align*}
and using the fact that $z_q = - z_{1-q}$ for all $q \in (0, 1)$, we obtain
\begin{align*}
	\mathbb{P}
	\left(
	\hat \theta_n \left(1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}\right) 
	\leq 
	\theta 
	\leq
	\hat \theta_n \left(1 + \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}} \right)
	\right) 
	=
	1 - \alpha
\end{align*}

So we can define the asymptotic confidence interval for $\theta$ by
\[
    C1_{1-\alpha} (\theta)
    =
    \left[
        \hat \theta_n \left(1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}\right) 
        ,
        \hat \theta_n \left(1 + \frac{z_1 - {\frac{\alpha}{2}}}{\sqrt{n}} \right)
    \right]
\]

\subsection{C3}

Use the Delta method with the function $g(\theta) = \log (\theta)$ and the knowledge from \eqref{MLE is normally distributed}.

\[
	g(\theta) \sim \mathcal N \left( g(\theta) , \left( g'(\theta) \right)^2 \mathcal I (\theta)^{-1} \right),
\]

so we can conclude that 

\[
	\log(\hat \theta_n) 
	\sim 
	\mathcal N \left( \log(\theta) , \left( \frac{1}{\theta^2} \frac{\theta^2}{n} \right) \right)
