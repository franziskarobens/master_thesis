\chapter{Confidence Intervals}

\begin{definition}[Confidence Interval]
Let $\mathbf X = (X_1, \dots, X_n)$ be a random sample with distribution depending on an unknown parameter $\theta \in \Theta$. A \emph{confidence interval} $CI$ for $\theta$ at confidence level $1 - \alpha$, $\alpha \in (0,1)$, 
is an interval
\[
	CI = [L(\mathbf X), U(\mathbf X)]
\]
where $L(\mathbf X)$ and $U(\mathbf X)$ are statistics satisfying $L(\mathbf X) \leq U(\mathbf X)$ and
\[
    \mathbb{P}_\theta \big( L(\mathbf X) \leq \theta \leq  U(\mathbf X) \big) \geq 1 - \alpha
\]
for all $\theta \in \Theta$.
\end{definition}




\begin{definition}[Lower Confidence Bound/ One-Sided Lower Confidence Interval]
	
The statistic $L(\mathbf X)$ is a lower $1 - \alpha$ confidence bound for $\theta$ if 
\[
	\mathbb{P}_\theta \big( L(\mathbf X) \leq \theta \big) \geq 1 - \alpha
\]
for all $\theta \in \Theta$.
The corresponding one-sided lower confidence interval is 
\[
	CI = [L(\mathbf X) , \infty )
\]
\end{definition}

Analogously, we can define a \emph{one-sided upper confidence interval}.

\begin{definition}[Upper Confidence Bound/ One-Sided Upper Confidence Interval]
	
	The statistic $U(\mathbf X)$ is an upper $1 - \alpha$ confidence bound for $\theta$ if 
	\[
	\mathbb{P}_\theta \big( \theta \leq  U(\mathbf X) \big) \geq 1 - \alpha
	\]
	for all $\theta \in \Theta$.
	The corresponding one-sided upper confidence interval is 
	\[
	CI = (- \infty, U(\mathbf X) ].
	\]
\end{definition}


If the parameter space $\Theta$ is a strict subset of $\mathbb{R}$, a one-sided confidence interval need not extend all the way to $\pm\infty$. Since $\theta \in \Theta$ is already known a priori, it is natural to intersect the interval with $\Theta$:
\[
CI_{\text{lower}} = \big[ L(\mathbf X),\, \infty \big) \cap \Theta
\qquad \text{or} \qquad
CI_{\text{upper}} = \big( -\infty,\, U(\mathbf X) \big] \cap \Theta,
\]
for the one-sided lower and upper confidence interval, respectively. This intersection does not affect the coverage guarantee: $\mathbb{P}_\theta\big(\theta \ge L(\mathbf X)\big) \ge 1-\alpha$ and $\mathbb{P}_\theta\big(\theta \le U(\mathbf X)\big) \ge 1-\alpha$ are only required to hold for $\theta \in \Theta$, so excluding values outside $\Theta$ removes no admissible outcome and hence leaves the respective coverage probability unchanged.

For example, if $\theta$ denotes the mean of an exponential distribution, so that $\Theta = (0, \infty)$, the one-sided upper confidence interval becomes
\[
CI = \big( 0,\, U(\mathbf X) \big].
\]



\section{Exact vs. Asymptotic CIs}

\begin{definition}[Exact confidence interval]
	A confidence interval $CI$ for a parameter $\theta$ is called \textbf{exact}, if and only if
	\[
		\mathbb{P}_{\theta} \left( \theta \in CI \right) = 1 - \alpha.
	\]
\end{definition}

\begin{definition}[Asymptotic confidence interval]
	A confidence interval $CI$ for a parameter $\theta$ is called \textbf{asymptotic}, if
	\[
	\mathbb{P}_{\theta} \left( \theta \in CI \right) \overset{n \to \infty}{\longrightarrow} 1 - \alpha.
	\]
	where $n$ is the sample size.
\end{definition}

So the main difference between exact and asymtotic confidence intervals is, that exact confidence intervals guarantee a confidence niveau of $1 - \alpha$ even for a small sample size $n$, but asymptotic confidence intervals only have confidence niveau $1 - \alpha$ in the limit $n \to \infty$.

	

\begin{example}[Exact vs. Asymptotic Confidence Interval]
	Let $X_1, \ldots , X_n \overset{iid}{\sim} \mathcal N (\mu, \sigma)$. Consider two cases.
	
	
	\textbf{Case 1: standard deviation $\sigma$ is known} \\
	In this case we can construct an exact confidence interval the following way.
	Observe that the arithmetic mean of the observations fulfils $\bar X = \frac{1}{n} \sum_{i=1}^n X_i \sim \mathcal N \left( \mu, \frac{\sigma^2}{n} \right)$ and by scaling this term, we get
	\[
		\frac{\sqrt{n}(\bar X - \mu)}{\sigma} \sim \mathcal N (0, 1).
	\]
	This distribution holds for every sample size $n$ so it is not only asymptotic. 
	Let $\alpha \in (0, 1)$ be arbitrary. We can construct a confidence interval by 
	\begin{align}
		& 1 - \alpha \notag \\
		=&
		1 - \frac{\alpha}{2} - \frac{\alpha}{2} \notag \\
		=&
		\mathbb{P} \left( z_{\frac{\alpha}{2}} \leq 
		\frac{\sqrt{n}(\bar X - \mu)}{\sigma}
		\leq z_{1 - \frac{\alpha}{2}} \right) \notag \\
		=&
		\mathbb{P} 
		\left( 
		\frac{\sigma}{\sqrt{n}} z_{\frac{\alpha}{2}} \leq \bar X - \mu \leq \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}		
		\right)  \notag \\
		=&
		\mathbb{P} 
		\left( 
		\bar X - \frac{\sigma}{\sqrt{n}} z_{\frac{\alpha}{2}} \geq  \mu \geq \bar X - \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}		
		\right)
	\end{align}
	For the quantiles of the normal distributino it holds $z_p = - z_{1-p}$ for all $p \in (0, 1)$, so we can substitute the quantile $z_{\frac{\alpha}{2}}$ above. The $1- \alpha$ confidence interval is given by
	\[
		CI_{1 - \alpha}^{\text{exact}} (\mu) 
		=
		\left[ 
		\bar X - \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		,
		\bar X + \frac{\sigma}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		\right]
	\]
	We have shown that the probability that the real parameter $\mu$ is in $CI_{1 - \alpha}^{\text{exact}} (\mu)$ is $1 - \alpha$ independently of the sample size $n$, so $CI_{1 - \alpha}^{\text{exact}} (\mu)$ is an exact confidence interval. 
	
	
	\textbf{Case 2: standard deviation $\sigma$ is unknown} \\
	
	
	From the Central Limit Theorem, we know that
	\[
		\frac{\sqrt{n} (\bar X - \mu)}{\sigma} \overset{n \rightarrow \infty}{\longrightarrow} \mathcal N (0, 1).
	\]
	Since $\sigma$ is unknown, we need to substitute it somehow.
	It holds 
	\[
		S_n := \sqrt{\frac{1}{n-1} \sum_{i=1}^n \left( X_i - \bar X \right)^2 }  \overset{p}{\longrightarrow} \sigma 
	\]
	\textcolor{red}{($\star$) Warum ist $S_n$ so verteilt?}
	
	With Slutsky's Theorem, we can conclude that 
	\[
		\frac{\sqrt{n} (\bar X - \mu)}{S_n} \overset{n \rightarrow \infty}{\longrightarrow} \mathcal N (0, 1).
	\]
	and with that knowledge we can create a $1 - \alpha$ confidence interval similarily to case 1
	\[
		CI_{1 - \alpha}^{\text{asymptotic}} (\mu) 
		=
		\left[ 
		\bar X - \frac{S_n}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		,
		\bar X + \frac{S_n}{\sqrt{n}} z_{1 - \frac{\alpha}{2}}
		\right].
	\]
	This confidence interval is only asymptotic because \textcolor{red}{($\star$) Warum nur asymptotic? Warum funktioniert die einsetzung mit slutsky nur asymptotic?}
	
	
	In Figure~\ref{fig:exact_vs_asymptotic}, we can see the non-coverage probability of the intervals $CI_{1 - \alpha}^{\text{exact}} (\mu)$ and $CI_{1 - \alpha}^{\text{asymptotic}} (\mu)$ dependent on the sample size. In Listing~\ref{listing:exact vs asymptotic} is the code used for generating Figure~\ref{fig:exact_vs_asymptotic}. 
	
	We can observe that \(CI_{1-\alpha}^{\mathrm{exact}}(\mu)\) maintains a non-coverage probability close to \(\alpha=0.1\) for all sample sizes $n$, while the non-coverage probability of \(CI_{1-\alpha}^{\mathrm{asymptotic}}(\mu)\) approaches \(\alpha\) as \(n\) increases.

For each simulation, a sample from \(N(\mu,\sigma^2)\) with \(\mu=10\) and \(\sigma=100\) is generated, and it is recorded whether \(\mu\) is contained in the respective confidence interval.}
		\label{fig:exact_vs_asymptotic}
	\end{figure}

\subsection{C1}

The MLE $\hat \theta$ of independent exponentially distributed random variables $X_1, \ldots , X_n$ is given by
\[
    \hat \theta_n = \frac{1}{\frac{1}{n} \sum_{i=1}^n  X_i}
\]

\[
    \hat \theta_n \overset{d}{\longrightarrow} \mathcal{N} (\theta, \mathcal I (\theta)^{-1})
\]

Replacing the variance by the observed variance:
\[
	\hat \theta_n \overset{d}{\longrightarrow} \mathcal{N} (\theta, \mathcal I (\hat \theta_n)^{-1})
\]

Note that this only holds if the sample size $n$ is very large.

Using this, we know
\[
    \sqrt{\mathcal I (\hat \theta_n )} \left(\hat \theta_n - \theta \right) \sim \mathcal N (0, 1)
\]


For example for $X_1, \ldots , X_n \overset{iid}{\sim} \text{Exp}(\theta)$, the Fisher information can be obtained by calculating the Likelihood function

\[
    L(\theta; \mathbf{x}) 
    =
    \prod_{i=1}^n  \theta \exp(- \theta x_i)
    =
    \theta^n \exp \left( - \theta \sum_{i=1}^n  x_i \right)
\]

\begin{align}
    l(\theta) &= n \log( \theta ) - \theta \sum_{i=1}^n  x_i  \label{log likelihood exp} \\
    l'(\theta) &= \frac{n}{\theta} - \sum_{i=1}^n  x_i \notag \\
    l''(\theta) &= - \frac{n}{\theta^2} \notag
\end{align}

So the Fisher information is 
\[
    \mathcal I (\theta) = \frac{n}{\theta^2}
\]
and therefore the MLE has distribution 

\[
    \hat \theta_n \sim \mathcal{N} \left(\theta , \frac{ \hat \theta_n^2}{n} \right)
\]

So we can get the standardized MLE 
\[
    \sqrt{n}\frac{\hat \theta_n - \theta}{\hat \theta_n} \sim \mathcal{N} \left(0 , 1 \right)
\]

With the knowledge of the distribution of the standardized MLE, one can construct an exact confidence interval. Notice that 
\begin{align*}
    \mathbb{P}
    \left(
    z_{\alpha /2} \leq \sqrt{n}\frac{\hat \theta_n - \theta}{\hat \theta_n} \leq z_{1 - \frac{\alpha}{2}}
    \right) 
    =
    1 - \alpha
\end{align*}
and by transforming the event one gets 
\begin{align*}
    \mathbb{P}
    \left(
        \hat \theta_n \left(1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}\right) 
    \leq 
        \theta 
    \leq
        \hat \theta_n \left(1 - \frac{z_{\frac{\alpha}{2}}}{\sqrt{n}} \right)
    \right) 
    =
    1 - \alpha
\end{align*}
and using the fact that $z_q = - z_{1-q}$ for all $q \in (0, 1)$, we obtain
\begin{align*}
	\mathbb{P}
	\left(
	\hat \theta_n \left(1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}\right) 
	\leq 
	\theta 
	\leq
	\hat \theta_n \left(1 + \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}} \right)
	\right) 
	=
	1 - \alpha
\end{align*}

So we can define the asymptotic confidence interval for $\theta$ by
\[
    C1_{1-\alpha} (\theta)
    =
    \left[
        \hat \theta_n \left(1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}\right) 
        ,
        \hat \theta_n \left(1 + \frac{z_1 - {\frac{\alpha}{2}}}{\sqrt{n}} \right)
    \right]
\]

\textcolor{red}{($\star$) Problem mit $N \in \{1, 2\}\}$ erklären}

For the CI C2, instead of substituting the variance $\mathcal I (\theta)^{-1}$ by the observed variance like in C1, we use the actual variance

\begin{equation} \label{MLE is normally distributed}
	\hat \theta_n \sim \mathcal N \left( \theta, \mathcal I (\theta)^{-1} \right)
\end{equation}

with $\mathcal I (\theta) = \frac{n}{\theta ^2}$
\[
	\mathbb{P} \left( 
		z_{\frac{\alpha}{2}}
	\leq
		\sqrt{n} \frac{\hat \theta_n - \theta}{\theta} 
	\leq 
		z_{1 - \frac{\alpha}{2}}
	 \right)
	 =
	 1 - \alpha
\]
and performing similar steps as for C1, we get

\[
	\mathbb{P} \left( 
			\frac{\hat \theta_n}{\frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}} + 1}
		\leq
			\theta
		\leq 
			\frac{\hat \theta_n}{- \frac{z_{ 1 - \frac{\alpha}{2}}}{\sqrt{n}} + 1}
	\right)
	=
	1 - \alpha
\]
so the confidence interval C2 is given by

\[
	C2_{1-\alpha} (\theta)
	=
	\left[
	\frac{\hat \theta_n}{1 + \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}} 
	,
	\frac{\hat \theta_n}{1 - \frac{z_{1 - \frac{\alpha}{2}}}{\sqrt{n}}}
	\right]
\]

For certain small sample sizes $n$, the interval $C2_{1-\alpha}(\theta)$ has a negative upper bound and a positive lower bound. In this case, it is not a valid interval. This can be seen in Figure~\ref{fig:sundberg_alpha_comparison}, which plots the sample size $n$ against the quantity $c_u - c_l$. This quantity is proportional to the width of the interval. The exact width is obtained by multiplying $c_u - c_l$ by $\hat\theta_n$.

The range of small $n$ for which $c_u - c_l$ is negative depends on $\alpha$. 
For $\alpha = 0.01$, the interval is invalid for $n \leq 6$, as shown in Figure~\ref{fig:sundberg_alpha_001}.
For $\alpha = 0.1$, the interval is invalid for $n \in \{0, 1, 2\}$, as shown in Figure~\ref{fig:sundberg_alpha_01}. 

It holds 
\[
	\frac{z_{1- \frac{\alpha}{2}}}{\sqrt{n}} > 1
	\quad \text{if and only if} \quad
	n < z_{1-\alpha/2}^2
\]
and exactly in this case, the denominator of the upper bound $1 - \frac {z_{1-\alpha/2}}{\sqrt{n}}$ is negative, while the lower bound is always positive for all $n \in \mathbb N _0$ and $\alpha \in (0, 1)$. 
So the confidence interval $C2_{1-\alpha}(\theta)$ is invalid if $n < z_{1 - \frac{\alpha}{2}}^2$.

The range of $n$ changes with $\alpha$ because the critical value $z_{1-\alpha/2}^2$ increases as $\alpha$ decreases. In other words, a smaller significance level leads to a larger critical value, which increases the range of sample sizes for which $c_u$ is negative.

For example, when $\alpha = 0.1$, the interval is invalid for all
\[
n < z_{0.95}^2 \approx 2.706.
\]
When $\alpha = 0.01$, the interval is invalid for all
\[
n < z_{0.995}^2 \approx 6.635.
\]
Thus, a smaller significance level results in a larger range of sample sizes for which the interval is invalid. This is consistent with the observations shown in Figure~\ref{fig:sundberg_alpha_comparison}.

Since $Y_1, \ldots , Y_n \overset{iid}{\sim} \text{Exp} \left( \frac{1}{\theta} \right)$,
